## Anti-Patterns to Flag

| Anti-pattern | Consequence and preferred pattern |
|-------------|-----------------------------------|
| `await response.text()` or similar buffering on unbounded data | Can exhaust Worker memory; [stream large or unbounded bodies](references/runtime-patterns.md#stream-request-and-response-bodies). |
| Hardcoded secrets in source or config | Leaks credentials through version control; use Wrangler secrets. |
| `Math.random()` for security-sensitive tokens or IDs | Predictable values; use `crypto.randomUUID()` or `crypto.getRandomValues()`. |
| Async work started without awaiting, returning, or attaching it to `ctx.waitUntil()` | Work can be dropped and errors missed; tie it to the request or background-work lifetime. |
| Module-level mutable request state | Leaks data across requests and can cause I/O ownership errors; pass request state explicitly. |

| `ctx.passThroughOnException()` used as general error handling | Can conceal Worker failures by forwarding to the origin; use explicit error handling and structured error responses. |
| Hand-written `Env` that duplicates Wrangler bindings | Can drift from configuration; generate binding types with `wrangler types`. |
| Direct string comparison of secret values | Can expose timing differences; use the [Web Crypto comparison pattern](references/runtime-patterns.md#use-web-crypto-for-secure-token-generation). |
| Destructuring `ctx` methods, such as `const { waitUntil } = ctx` | Loses the receiver; call `ctx.waitUntil(...)`. |
| `any` on `Env` or handler parameters | Hides binding and handler contract errors; use the project's generated and platform types. |
| `as unknown as T` to force a platform type match | Hides incompatibilities; fix the underlying contract. |
| `implements` used in place of extending a platform base class | Does not inherit runtime behavior, `this.ctx`, or `this.env`; use the appropriate base class. |
| Unbound `env.X` in a platform class method | Bindings are available through `this.env.X`; see [binding access patterns](references/platform-apis.md#binding-access--the-most-common-error). |
| Applying one serialization rule across Queues, Workflow steps, storage, and WebSockets | Can reject valid payloads or accept unsupported ones; check the [specific API and encoding](references/platform-apis.md#serialization-boundaries). |

/**
 * Rewrites the KV record of every hostname of a site from D1, the source of truth: status,
 * published pointer and redirect. One writer per site at a time (D1 lease). A caller that
 * cannot take the lease leaves the site marked unsynced; the holder re-reads D1 after writing
 * and catches up, and the reconcile cron covers anything left. Returns whether KV now matches.
 */
export async function syncHostRecords(env: Env, siteId: string): Promise<boolean> {
  await clearRoutesSynced(env.DB, siteId);
  const token = await acquireRoutesLease(env.DB, siteId);
  if (!token) return false;
  try {
    for (let attempt = 0; attempt < 3; attempt++) {
      const state = await snapshot(env, siteId);
      for (const domain of state.domains) {
        // Fence: a holder whose lease expired must not overwrite a newer writer.
        if (!(await holdsRoutesLease(env.DB, siteId, token))) return false;
        // Custom hostnames that are not active yet must not serve the site.
        if (domain.kind === "custom" && domain.status !== "active") {
          await kvWrite(() => env.ROUTES.delete(kvHostKey(domain.hostname)));
          continue;
        }
        const record: HostRecord = {
          siteId: state.site.id,
          status: state.status,
          pointer: state.pointer,
          redirectTo: domain.hostname === state.primary ? null : state.primary,
        };
        await kvWrite(() => env.ROUTES.put(kvHostKey(domain.hostname), JSON.stringify(record)));
      }
      if ((await snapshot(env, siteId)).key === state.key) {
        await markRoutesSynced(env.DB, siteId, state.site.updated_at);
        return true;
      }
    }
    return false;
  } finally {
    await releaseRoutesLease(env.DB, siteId, token);
  }
}

/**
 * For request handlers: syncs now, retrying briefly while another writer holds the lease, so
 * the caller's change is live when it responds. The cron catches anything still pending.
 */
export async function syncHostRecordsNow(env: Env, siteId: string) {
  for (let attempt = 0; attempt < 4; attempt++) {
    if (await syncHostRecords(env, siteId)) return true;
    await sleep(1000);
  }
  return false;
}

/**
 * Cron safety net: re-syncs sites whose host records may lag D1 (a request failed after D1
 * committed, or KV kept rejecting writes). Guarantees serving converges to D1 within minutes.
 */
export async function reconcileHostRecords(env: Env) {
  const ids = await sitesNeedingRouteSync(env.DB);
  let synced = 0;
  for (const id of ids) {
    try {
      if (await syncHostRecords(env, id)) synced++;
    } catch (error) {
      console.error("[routes] reconcile failed", id, error);
    }
  }
  return { pending: ids.length, synced };
}

const LEASE_MS = 60_000;

/** Takes the per-site route-sync lease; null when another writer holds it. */
export async function acquireRoutesLease(db: D1Database, siteId: string): Promise<string | null> {
  const token = crypto.randomUUID();
  const now = new Date();
  const result = await db
    .prepare(
      "UPDATE sites SET routes_lock_token = ?, routes_lock_until = ? WHERE id = ? AND (routes_lock_until IS NULL OR routes_lock_until < ?)",
    )
    .bind(token, new Date(now.getTime() + LEASE_MS).toISOString(), siteId, now.toISOString())
    .run();
  return result.meta.changes === 1 ? token : null;
}

export async function holdsRoutesLease(db: D1Database, siteId: string, token: string) {
  const row = await db
    .prepare("SELECT 1 AS held FROM sites WHERE id = ? AND routes_lock_token = ? AND routes_lock_until >= ?")
    .bind(siteId, token, new Date().toISOString())
    .first<{ held: number }>();
  return Boolean(row);
}

export async function releaseRoutesLease(db: D1Database, siteId: string, token: string) {
  await db
    .prepare("UPDATE sites SET routes_lock_token = NULL, routes_lock_until = NULL WHERE id = ? AND routes_lock_token = ?")
    .bind(siteId, token)
    .run();
}

/**
 * Renders every page of a version to R2 unless it is already rendered against the current
 * shared assets and origin.
 */
async function ensureRendered(env: Env, version: VersionRow, origin: string) {
  const config = JSON.parse(version.config) as SiteConfig;
  if (version.rendered_at && version.asset_version === ASSET_VERSION && version.rendered_origin === origin) {
    return routesOf(config);
  }
  const { pages } = await renderSite(config, { assetBase: assetBase(), origin });
  for (const page of pages) {
    await env.SITES.put(r2PageKey(version.site_id, version.id, page.path), page.html, {
      httpMetadata: { contentType: "text/html; charset=utf-8" },
    });
  }
  await markRendered(env.DB, version.id, ASSET_VERSION, origin);
  return pages.map((page) => page.route);
}

async function pointTo(env: Env, site: SiteRow, version: VersionRow, pages: string[]) {
  const publishedAt = await markPublished(env.DB, site.id, version.id, ASSET_VERSION);
  const pointer: PublishedPointer = { versionId: version.id, number: version.number, publishedAt, pages };
  await env.ROUTES.put(kvSiteKey(site.id), JSON.stringify(pointer));
  await syncHostRecordsNow(env, site.id);
  return pointer;
}

/** Publish = render (if needed) → R2 → move the KV pointer. Visible worldwide within about a minute. */
export async function publish(env: Env, site: SiteRow, versionId?: string) {
  await requireSharedAssets(env);
  const version = versionId ? await getVersion(env.DB, site.id, versionId) : await latestVersion(env.DB, site.id);
  const pages = await ensureRendered(env, version, await siteOrigin(env, site));
  return pointTo(env, site, version, pages);
}

/** Rollback = move the pointer back to a version that was published before. No re-render. */
export async function rollback(env: Env, site: SiteRow, versionId: string) {
  const version = await getVersion(env.DB, site.id, versionId);
  if (!version.first_published_at) throw new HttpError(409, "게시된 적 없는 버전으로는 롤백할 수 없습니다");
  const exists = await env.SITES.head(r2PageKey(site.id, version.id));
  if (!exists) throw new HttpError(409, "렌더된 파일이 R2에 없습니다. publish로 다시 게시하세요");
  return pointTo(env, site, version, routesOf(JSON.parse(version.config) as SiteConfig));
}

async function serveImmutable(env: Env, key: string) {
  const object = await env.SITES.get(key);
  if (!object) return new Response("not found", { status: 404 });
  const headers = new Headers();
  object.writeHttpMetadata(headers);
  headers.set("cache-control", "public, max-age=31536000, immutable");

async function handle(request: Request, env: Env): Promise<Response> {

  if (method === "GET" && url.pathname === "/health") return json({ ok: true });

  // Shared assets are public and immutable; previews load them from here.

  // Uploaded photos (public, content-addressed) so previews show them before publishing.
  const media = method === "GET" ? MEDIA_PATH.exec(url.pathname) : null;
  if (media) return serveImmutable(env, r2MediaKey(media[1]!, media[2]!));

  // Preview: anyone with the token can view a version. Never indexed. /preview/<token>[/privacy]
  if (method === "GET" && parts[0] === "preview" && (parts.length === 2 || parts.length === 3)) {
    const version = await versionByPreviewToken(env.DB, parts[1]!);
    const site = await getSite(env.DB, version.site_id);
    const { pages } = await renderSite(JSON.parse(version.config) as SiteConfig, {
      assetBase: assetBase(),
      origin: await siteOrigin(env, site),
      preview: true,
      // Site links are root-relative (`/#inquiry`, `/privacy`); keep them inside this preview.
      linkPrefix: `/preview/${parts[1]}`,
    });
    const route = parts[2] ? `/${parts[2]}` : "/";
    const page = pages.find((entry) => entry.route === route);
    if (!page) throw new HttpError(404, "없는 페이지입니다");
    return new Response(page.html, {
      headers: {
        "content-type": "text/html; charset=utf-8",
        "x-robots-tag": "noindex, nofollow",
        "cache-control": "no-store",
      },
    });
  }

  requireAdmin(request, env.ADMIN_TOKEN);

  if (method === "PUT" && parts[0] === "admin" && parts[1] === "shared" && parts.length > 3) {

    });

    const result = await addVersion(env.DB, site, validated(body.config), creator(body.createdBy), body.note ?? null);
    return json(result, 201);
  }

  if (parts[2] === "assets" && parts.length === 3) {
    if (method === "POST") return json(await uploadAsset(env, site, request), 201);

  }

    60	/**
    61	 * Rewrites the KV record of every hostname of a site from D1, the source of truth: status,
    62	 * published pointer and redirect. One writer per site at a time (D1 lease). A caller that
    63	 * cannot take the lease leaves the site marked unsynced; the holder re-reads D1 after writing
    64	 * and catches up, and the reconcile cron covers anything left. Returns whether KV now matches.
    65	 */
    66	export async function syncHostRecords(env: Env, siteId: string): Promise<boolean> {
    67	  await clearRoutesSynced(env.DB, siteId);
    68	  const token = await acquireRoutesLease(env.DB, siteId);
    69	  if (!token) return false;
    70	  try {
    71	    for (let attempt = 0; attempt < 3; attempt++) {
    72	      const state = await snapshot(env, siteId);
    73	      for (const domain of state.domains) {
    74	        // Fence: a holder whose lease expired must not overwrite a newer writer.
    75	        if (!(await holdsRoutesLease(env.DB, siteId, token))) return false;
    76	        // Custom hostnames that are not active yet must not serve the site.
    77	        if (domain.kind === "custom" && domain.status !== "active") {
    78	          await kvWrite(() => env.ROUTES.delete(kvHostKey(domain.hostname)));
    79	          continue;
    80	        }
    81	        const record: HostRecord = {
    82	          siteId: state.site.id,
    83	          status: state.status,
    84	          pointer: state.pointer,
    85	          redirectTo: domain.hostname === state.primary ? null : state.primary,
    86	        };
    87	        await kvWrite(() => env.ROUTES.put(kvHostKey(domain.hostname), JSON.stringify(record)));
    88	      }
    89	      if ((await snapshot(env, siteId)).key === state.key) {
    90	        await markRoutesSynced(env.DB, siteId, state.site.updated_at);
    91	        return true;
    92	      }
    93	    }
    94	    return false;
    95	  } finally {
    96	    await releaseRoutesLease(env.DB, siteId, token);
    97	  }
    98	}

   101	 * For request handlers: syncs now, retrying briefly while another writer holds the lease, so
   102	 * the caller's change is live when it responds. The cron catches anything still pending.
   103	 */
   104	export async function syncHostRecordsNow(env: Env, siteId: string) {
   105	  for (let attempt = 0; attempt < 4; attempt++) {
   106	    if (await syncHostRecords(env, siteId)) return true;
   107	    await sleep(1000);
   108	  }
   109	  return false;
