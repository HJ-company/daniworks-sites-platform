/** Public status of a site as the router sees it. */
export function hostStatus(site: SiteRow): HostRecord["status"] {
  if (site.cancelled_at) return "cancelled";
  return site.status === "suspended" ? "suspended" : "active";

/** Routes a stored config renders to (v2 adds the privacy page). */
export const routesOf = (config: { version?: unknown }) => (config.version === 2 ? ["/", "/privacy"] : ["/"]);

async function snapshot(env: Env, siteId: string) {
  const { site, version } = await servingState(env.DB, siteId);
  const pointer: PublishedPointer | null =
    version && site.published_at
      ? {
          versionId: version.id,
          number: version.number,
          publishedAt: site.published_at,
          pages: routesOf(JSON.parse(version.config) as { version?: unknown }),
        }
      : null;
  const primary = await primaryHostname(env.DB, site, env.BASE_DOMAIN);
  const { results: domains } = await env.DB.prepare("SELECT hostname, kind, status FROM domains WHERE site_id = ?")
    .bind(site.id)
    .all<{ hostname: string; kind: string; status: string }>();
  const status = hostStatus(site);
  const key = JSON.stringify([status, pointer?.versionId, pointer?.publishedAt, primary, domains]);
  return { site, pointer, primary, domains, status, key };
}

/**
 * Rewrites the KV record of every hostname of a site from D1, the source of truth: status,
 * published pointer and redirect. Never trusts a caller's row or a KV read (eventually
 * consistent). After writing it re-reads D1 and rewrites if a concurrent publish, rollback or
 * status change landed in between, so the last writer always reflects the latest D1 state.
 */
export async function syncHostRecords(env: Env, siteId: string) {
  for (let attempt = 0; attempt < 3; attempt++) {
    const state = await snapshot(env, siteId);
    for (const domain of state.domains) {
      // Custom hostnames that are not active yet must not serve the site.
      if (domain.kind === "custom" && domain.status !== "active") {
        await env.ROUTES.delete(kvHostKey(domain.hostname));
        continue;
      }
      const record: HostRecord = {
        siteId: state.site.id,
        status: state.status,
        pointer: state.pointer,
        redirectTo: domain.hostname === state.primary ? null : state.primary,
      };
      await env.ROUTES.put(kvHostKey(domain.hostname), JSON.stringify(record));
    }
    if ((await snapshot(env, siteId)).key === state.key) return;
  }
}

async function requireSharedAssets(env: Env) {
  const head = await env.SITES.head(r2SharedKey(`${ASSET_VERSION}/_assets/site.css`));
  if (!head) {
    throw new HttpError(409, `공용 에셋 ${ASSET_VERSION}이 R2에 없습니다. scripts/upload-shared.mjs를 먼저 실행하세요`);
  }
}

/**
 * Renders every page of a version to R2 unless it is already rendered against the current
 * shared assets and origin.
 */
async function ensureRendered(env: Env, version: VersionRow, origin: string) {
  const config = JSON.parse(version.config) as SiteConfig;
  if (version.rendered_at && version.asset_version === ASSET_VERSION && version.rendered_origin === origin) {
    return routesOf(config);
  }
  const { pages } = await renderSite(config, { assetBase: assetBase(), origin });
  for (const page of pages) {
    await env.SITES.put(r2PageKey(version.site_id, version.id, page.path), page.html, {
      httpMetadata: { contentType: "text/html; charset=utf-8" },
    });
  }
  await markRendered(env.DB, version.id, ASSET_VERSION, origin);
  return pages.map((page) => page.route);
}

async function pointTo(env: Env, site: SiteRow, version: VersionRow, pages: string[]) {
  const publishedAt = await markPublished(env.DB, site.id, version.id, ASSET_VERSION);
  const pointer: PublishedPointer = { versionId: version.id, number: version.number, publishedAt, pages };
  await env.ROUTES.put(kvSiteKey(site.id), JSON.stringify(pointer));
  await syncHostRecords(env, site.id);
  return pointer;
}

/** Publish = render (if needed) → R2 → move the KV pointer. Visible worldwide within about a minute. */
export async function publish(env: Env, site: SiteRow, versionId?: string) {
  await requireSharedAssets(env);
  const version = versionId ? await getVersion(env.DB, site.id, versionId) : await latestVersion(env.DB, site.id);
  const pages = await ensureRendered(env, version, await siteOrigin(env, site));
  return pointTo(env, site, version, pages);
}

/** Rollback = move the pointer back to a version that was published before. No re-render. */
export async function rollback(env: Env, site: SiteRow, versionId: string) {
  const version = await getVersion(env.DB, site.id, versionId);
  if (!version.first_published_at) throw new HttpError(409, "게시된 적 없는 버전으로는 롤백할 수 없습니다");
  const exists = await env.SITES.head(r2PageKey(site.id, version.id));
  if (!exists) throw new HttpError(409, "렌더된 파일이 R2에 없습니다. publish로 다시 게시하세요");
  return pointTo(env, site, version, routesOf(JSON.parse(version.config) as SiteConfig));
}

## Anti-Patterns to Flag

| Anti-pattern | Consequence and preferred pattern |
|-------------|-----------------------------------|
| `await response.text()` or similar buffering on unbounded data | Can exhaust Worker memory; [stream large or unbounded bodies](references/runtime-patterns.md#stream-request-and-response-bodies). |
| Hardcoded secrets in source or config | Leaks credentials through version control; use Wrangler secrets. |
| `Math.random()` for security-sensitive tokens or IDs | Predictable values; use `crypto.randomUUID()` or `crypto.getRandomValues()`. |
| Async work started without awaiting, returning, or attaching it to `ctx.waitUntil()` | Work can be dropped and errors missed; tie it to the request or background-work lifetime. |
| Module-level mutable request state | Leaks data across requests and can cause I/O ownership errors; pass request state explicitly. |
| Cloudflare REST API calls for operations available through Worker bindings | Adds network and authentication overhead; use the available binding. |
| `ctx.passThroughOnException()` used as general error handling | Can conceal Worker failures by forwarding to the origin; use explicit error handling and structured error responses. |
| Hand-written `Env` that duplicates Wrangler bindings | Can drift from configuration; generate binding types with `wrangler types`. |
| Direct string comparison of secret values | Can expose timing differences; use the [Web Crypto comparison pattern](references/runtime-patterns.md#use-web-crypto-for-secure-token-generation). |
| Destructuring `ctx` methods, such as `const { waitUntil } = ctx` | Loses the receiver; call `ctx.waitUntil(...)`. |
| `any` on `Env` or handler parameters | Hides binding and handler contract errors; use the project's generated and platform types. |
| `as unknown as T` to force a platform type match | Hides incompatibilities; fix the underlying contract. |
| `implements` used in place of extending a platform base class | Does not inherit runtime behavior, `this.ctx`, or `this.env`; use the appropriate base class. |
| Unbound `env.X` in a platform class method | Bindings are available through `this.env.X`; see [binding access patterns](references/platform-apis.md#binding-access--the-most-common-error). |
| Applying one serialization rule across Queues, Workflow steps, storage, and WebSockets | Can reject valid payloads or accept unsupported ones; check the [specific API and encoding](references/platform-apis.md#serialization-boundaries). |

export async function createSite(db: D1Database, config: SiteConfig, createdBy: CreatedBy, hostname: string) {
  const exists = await db.prepare("SELECT id FROM sites WHERE slug = ?").bind(config.slug).first();
  if (exists) throw new HttpError(409, `이미 쓰는 slug입니다: ${config.slug}`);
  const siteId = crypto.randomUUID();
  const versionId = crypto.randomUUID();
  const at = now();
  await db.batch([
    db
      .prepare("INSERT INTO sites (id, slug, template_id, created_at, updated_at) VALUES (?, ?, ?, ?, ?)")
      .bind(siteId, config.slug, config.templateId, at, at),
    db
      .prepare(
        "INSERT INTO site_versions (id, site_id, number, config, created_by, note, preview_token, created_at) VALUES (?, ?, 1, ?, ?, ?, ?, ?)",
      )
      .bind(versionId, siteId, JSON.stringify(config), createdBy, "최초 생성", token(), at),
    db
      .prepare("INSERT INTO domains (hostname, site_id, kind, created_at) VALUES (?, ?, 'subdomain', ?)")
      .bind(hostname, siteId, at),
  ]);
  return { siteId, versionId };
}

export async function addVersion(
  db: D1Database,
  site: SiteRow,
  config: SiteConfig,
  createdBy: CreatedBy,
  note: string | null,
) {
  if (config.slug !== site.slug) throw new HttpError(422, "slug는 바꿀 수 없습니다", [`slug: ${site.slug}`]);
  const latest = await latestVersion(db, site.id);
  const versionId = crypto.randomUUID();
  const at = now();
  await db.batch([
    db
      .prepare(
        "INSERT INTO site_versions (id, site_id, number, config, created_by, note, preview_token, created_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
      )
      .bind(versionId, site.id, latest.number + 1, JSON.stringify(config), createdBy, note, token(), at),
    db.prepare("UPDATE sites SET updated_at = ? WHERE id = ?").bind(at, site.id),
  ]);
  return { versionId, number: latest.number + 1 };
}

  await db.batch([
    db
      .prepare("UPDATE site_versions SET first_published_at = COALESCE(first_published_at, ?) WHERE id = ?")
      .bind(at, versionId),
    db
      .prepare(
        "UPDATE sites SET published_version_id = ?, published_at = ?, template_version = ?, updated_at = ? WHERE id = ?",
      )
      .bind(versionId, at, templateVersion, at, siteId),

  return at;
}

/** Parsed config with defaults filled; what gets stored. */
function validated(input: unknown): SiteConfig {
  const result = parseSiteConfig(input);
  if (!result.ok) throw new HttpError(422, "설정 검증 실패", result.errors);
  return result.config;

function creator(value: unknown): CreatedBy {
  return CREATORS.includes(value as CreatedBy) ? (value as CreatedBy) : "operator";

async function serveImmutable(env: Env, key: string) {
  const object = await env.SITES.get(key);
  if (!object) return new Response("not found", { status: 404 });
  const headers = new Headers();
  object.writeHttpMetadata(headers);
  headers.set("cache-control", "public, max-age=31536000, immutable");
  return new Response(object.body, { headers });
}

async function handle(request: Request, env: Env): Promise<Response> {
  const url = new URL(request.url);
  const parts = url.pathname.split("/").filter(Boolean);
  const method = request.method;

  if (method === "GET" && url.pathname === "/health") return json({ ok: true });

  // Shared assets are public and immutable; previews load them from here.
  if (method === "GET" && parts[0] === "_shared") return serveImmutable(env, r2SharedKey(parts.slice(1).join("/")));
  // Uploaded photos (public, content-addressed) so previews show them before publishing.
  const media = method === "GET" ? MEDIA_PATH.exec(url.pathname) : null;
  if (media) return serveImmutable(env, r2MediaKey(media[1]!, media[2]!));

  // Preview: anyone with the token can view a version. Never indexed. /preview/<token>[/privacy]
  if (method === "GET" && parts[0] === "preview" && (parts.length === 2 || parts.length === 3)) {
    const version = await versionByPreviewToken(env.DB, parts[1]!);
    const site = await getSite(env.DB, version.site_id);
    const { pages } = await renderSite(JSON.parse(version.config) as SiteConfig, {
      assetBase: assetBase(),
      origin: await siteOrigin(env, site),
      preview: true,
      // Site links are root-relative (`/#inquiry`, `/privacy`); keep them inside this preview.
      linkPrefix: `/preview/${parts[1]}`,
    });
    const route = parts[2] ? `/${parts[2]}` : "/";
    const page = pages.find((entry) => entry.route === route);
    if (!page) throw new HttpError(404, "없는 페이지입니다");
    return new Response(page.html, {
      headers: {
        "content-type": "text/html; charset=utf-8",
        "x-robots-tag": "noindex, nofollow",
        "cache-control": "no-store",
      },
    });
  }

  requireAdmin(request, env.ADMIN_TOKEN);

  // PUT /admin/shared/<version>/<path>: upload one shared asset file.
  if (method === "PUT" && parts[0] === "admin" && parts[1] === "shared" && parts.length > 3) {
    const key = r2SharedKey(parts.slice(2).join("/"));
    await env.SITES.put(key, request.body, {
      httpMetadata: { contentType: request.headers.get("content-type") ?? "application/octet-stream" },
    });
    return json({ key }, 201);
  }

  if (parts[0] !== "sites") throw new HttpError(404, "없는 경로입니다");

  // POST /sites {config, createdBy}
  if (method === "POST" && parts.length === 1) {
    const body = await readJson<{ config: unknown; createdBy?: CreatedBy }>(request);
    const config = validated(body.config);
    const hostname = `${config.slug}.${env.BASE_DOMAIN}`;
    const { siteId, versionId } = await createSite(env.DB, config, creator(body.createdBy), hostname);
    await syncHostRecords(env, siteId);
    return json({ siteId, versionId, hostname }, 201);
  }

  const site = await getSite(env.DB, parts[1]!);

  if (method === "GET" && parts.length === 2) {
    return json({ site, versions: await listVersions(env.DB, site.id) });
  }

  // POST /sites/:id/versions {config, createdBy, note}: new draft version
  if (method === "POST" && parts[2] === "versions") {
    const body = await readJson<{ config: unknown; createdBy?: CreatedBy; note?: string }>(request);
    const result = await addVersion(env.DB, site, validated(body.config), creator(body.createdBy), body.note ?? null);
    return json(result, 201);
  }

  // POST /sites/:id/assets: raw image body, or JSON {url}. GET lists them.
  if (parts[2] === "assets" && parts.length === 3) {
    if (method === "POST") return json(await uploadAsset(env, site, request), 201);
    if (method === "GET") return json({ assets: await listAssets(env, site) });
  }

  // POST /sites/:id/publish {versionId?}: latest version when omitted
  if (method === "POST" && parts[2] === "publish") {
    const body = await readJson<{ versionId?: string }>(request).catch(() => ({}) as { versionId?: string });
    return json(await publish(env, site, body.versionId));
  }

  // POST /sites/:id/suspend | /resume: stop or restart public serving (non-payment, requests).
  if (method === "POST" && (parts[2] === "suspend" || parts[2] === "resume") && parts.length === 3) {
    if (site.cancelled_at) throw new HttpError(409, "해지된 사이트입니다");
    const status = parts[2] === "suspend" ? "suspended" : "active";
    await setSiteStatus(env.DB, site.id, status);
    await syncHostRecords(env, site.id);
    return json({ siteId: site.id, status });
  }

  // POST /sites/:id/rollback {versionId}
  if (method === "POST" && parts[2] === "rollback") {
    const body = await readJson<{ versionId: string }>(request);
    if (!body.versionId) throw new HttpError(400, "versionId가 필요합니다");
    return json(await rollback(env, site, body.versionId));
  }

  throw new HttpError(404, "없는 경로입니다");
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {

      console.error(error);
      return json({ error: "서버 오류" }, 500);
    }
  },
