+export function lcpPreload(body: string): string {
+  const img = body.match(/<img\b[^>]*fetchPriority="high"[^>]*>|<img\b[^>]*fetchpriority="high"[^>]*>/i)?.[0];
+  if (!img) return "";
+  const attr = (name: string) => img.match(new RegExp(`\\s${name}="([^"]*)"`, "i"))?.[1];
+  const src = attr("src");
+  if (!src) return "";

+  return `<link rel="preload" as="image" href="${src}"${srcset ? ` imagesrcset="${srcset}"` : ""}${sizes ? ` imagesizes="${sizes}"` : ""} fetchpriority="high">`;

/** Request to the router with a Host header (Node cannot resolve *.localhost). */
export function routerRequest(host, path = "/", { method = "GET", body, headers = {} } = {}) {
  return new Promise((resolve, reject) => {
    const req = http.request(
      { host: "127.0.0.1", port: ROUTER_PORT, path, method, headers: { host: `${host}:${ROUTER_PORT}`, ...headers } },
      (res) => {
        let text = "";
        res.setEncoding("utf8");
        res.on("data", (c) => (text += c));
        res.on("end", () => resolve({ status: res.statusCode, headers: res.headers, body: text }));
      },
    );
    req.on("error", reject);
    if (body !== undefined) req.write(typeof body === "string" ? body : JSON.stringify(body));
    req.end();
  });
}

/**
 * Opens a page, scrolls it to trigger lazy images and reports hydration, console errors,
 * failed and external requests. `allowExternal` lists URL prefixes that may be requested.
 */
export async function inspectPage(browser, url, viewport, { screenshot, allowExternal = [] } = {}) {
  const page = await browser.newPage({ viewport });
  const origin = new URL(url).origin;
  const errors = [];
  const failed = [];
  const external = [];
  page.on("console", (m) => m.type() === "error" && errors.push(m.text()));
  page.on("pageerror", (e) => errors.push(String(e)));
  page.on("requestfailed", (r) => failed.push(r.url()));
  page.on("request", (r) => {
    const u = r.url();
    if (!u.startsWith(origin) && !u.startsWith("data:") && !allowExternal.some((p) => u.startsWith(p))) external.push(u);
  });
  await page.goto(url, { waitUntil: "networkidle" });
  await page.evaluate(async () => {
    for (let y = 0; y < document.body.scrollHeight; y += 400) {
      window.scrollTo(0, y);
      await new Promise((r) => setTimeout(r, 80));
    }
    window.scrollTo(0, 0);
  });
  await page.waitForLoadState("networkidle");
  const state = await page.evaluate(() => {
    const root = document.getElementById("root");
    return {
      hydrated:
        !!root &&
        Object.keys(root).some((k) => k.startsWith("__reactContainer")) &&
        [...root.querySelectorAll("*")].some((el) => Object.keys(el).some((k) => k.startsWith("__reactFiber"))),
      unloadedImages: [...document.images].filter((i) => !i.complete || i.naturalWidth === 0).map((i) => i.src),
    };
  });
  if (screenshot) await page.screenshot({ path: screenshot, fullPage: true });
  return { page, errors, failed, external, ...state };
}

/**
 * Lighthouse (mobile, simulated throttling), median of `runs` by performance score as
 * Lighthouse recommends: single runs vary with host CPU load (TBT especially).
 */
export async function lighthouse(url, runs = 3) {
  const all = [];
  for (let i = 0; i < runs; i++) all.push(await lighthouseOnce(url));
  const ok = all.filter((r) => !r.error).sort((a, b) => a.performance - b.performance);
  const median = ok[Math.floor(ok.length / 2)] ?? all[0];
  return { ...median, runs: all.map((r) => r.error ?? r.performance) };
}

async function lighthouseOnce(url) {
  const { default: run } = await import("lighthouse");
  const chromeLauncher = await import("chrome-launcher");
  const chrome = await chromeLauncher.launch({
    chromePath: chromium().executablePath(),
    chromeFlags: ["--headless=new", "--no-sandbox", "--disable-gpu", "--disable-dev-shm-usage"],
  });
  try {
    const result = await run(url, {
      port: chrome.port,
      output: "json",
      logLevel: "error",
      onlyCategories: ["performance", "accessibility", "best-practices", "seo"],
    });
    const lhr = result.lhr;
    if (lhr.runtimeError) return { error: lhr.runtimeError.code, performance: 0, report: lhr };
    const score = (c) => Math.round((lhr.categories[c]?.score ?? 0) * 100);
    return {
      performance: score("performance"),
      accessibility: score("accessibility"),
      bestPractices: score("best-practices"),
      seo: score("seo"),
      lcpMs: Math.round(lhr.audits["largest-contentful-paint"]?.numericValue ?? 0),
      cls: Number((lhr.audits["cumulative-layout-shift"]?.numericValue ?? 0).toFixed(3)),
      tbtMs: Math.round(lhr.audits["total-blocking-time"]?.numericValue ?? 0),
      report: lhr,
    };
  } finally {
    await chrome.kill();
  }
}

/** Reads or writes the router's local KV namespace (Docker: shared /state volume). */
export function localKv(action, key, value) {
  const args = ["exec", "wrangler", "kv", "key", action, key, ...(value !== undefined ? [value] : []), "--binding", "ROUTES", "--local", "--persist-to", process.env.STATE_DIR ?? "/state"];
  if (action === "get") args.push("--text");
  try {
    const out = execFileSync("pnpm", args, { cwd: "apps/router", encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] }).trim();
    // wrangler prints this and exits 0 for a missing key.
    return action === "get" && out === "Value not found" ? "" : out;
  } catch (error) {
    // A missing key exits non-zero; reads report it as empty.
    if (action === "get" && /not found/i.test(String(error.stderr ?? error.stdout ?? ""))) return "";
    throw error;
  }
}

export interface RenderOptions {
  /** Base URL of shared, template-versioned assets (CSS, client JS, template images). */
  assetBase: string;
  /** Public origin (`https://jefes.dsites.kr`) for canonical, OG and structured data URLs. */
  origin?: string;
  /** Skip the client bundle to measure server-only output. */
  hydrate?: boolean;
  /** Preview renders: noindex, no canonical, inquiry form disabled. */
  preview?: boolean;
  /** Prefix for root-relative site links (previews: `/preview/<token>`). */
  linkPrefix?: string;
  /** Turnstile frame the inquiry form embeds (`https://api.dsites.kr/turnstile-frame`). */
  turnstileFrame?: string | null;
}

export interface RenderedPage {
  route: string;
  /** Object path below the version prefix: `index.html`, `privacy/index.html`. */
  path: string;
  html: string;
}

export interface RenderResult {
  /** Home page HTML. */
  html: string;
  pages: RenderedPage[];
  ms: number;
}

const DATA_ID = "__SITE_DATA__";

function escapeJson(value: unknown) {
  return JSON.stringify(value).replace(/</g, "\\u003c");
}

function escapeHtml(value: string) {
  return value.replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" })[c]!);
}

/**
 * Preload for the LCP image, copied from the rendered `<img fetchpriority="high">` itself so
 * src, srcset and sizes always match what the image will pick (no double download).
 */
export function lcpPreload(body: string): string {
  const img = body.match(/<img\b[^>]*fetchPriority="high"[^>]*>|<img\b[^>]*fetchpriority="high"[^>]*>/i)?.[0];
  if (!img) return "";
  const attr = (name: string) => img.match(new RegExp(`\\s${name}="([^"]*)"`, "i"))?.[1];
  const src = attr("src");
  if (!src) return "";
  const srcset = attr("srcSet") ?? attr("srcset");
  const sizes = attr("sizes");
  return `<link rel="preload" as="image" href="${src}"${srcset ? ` imagesrcset="${srcset}"` : ""}${sizes ? ` imagesizes="${sizes}"` : ""} fetchpriority="high">`;
}

const absolute = (origin: string | undefined, url: string | undefined | null) =>
  !url ? undefined : /^https?:\/\//.test(url) ? url : origin ? `${origin}${url}` : undefined;

/** LocalBusiness (subtype by preset) and FAQPage. AI answers and search both read these. */
function structuredData(config: SiteConfig, origin: string | undefined, heroImage: string | undefined) {
  if (!isServiceConfig(config)) return [];
  const { business, content } = config;
  const org: Record<string, unknown> = {
    "@context": "https://schema.org",
    "@type": business.schemaType ?? presets[config.preset].schemaType,
    name: business.name,
    description: config.meta.description,
    telephone: business.phone,
    url: origin ? `${origin}/` : undefined,
    image: heroImage,
    address: business.address,
    areaServed: business.serviceAreas.length ? business.serviceAreas : undefined,
    sameAs: [business.blogUrl, business.kakaoUrl].filter(Boolean),
    makesOffer: content.services.items.map((item) => ({
      "@type": "Offer",
      itemOffered: { "@type": "Service", name: item.label, description: item.description },
    })),
  };
  const result: unknown[] = [JSON.parse(JSON.stringify(org))];
  if (content.faqs.length) {
    result.push({
      "@context": "https://schema.org",
      "@type": "FAQPage",
      mainEntity: content.faqs.map((faq) => ({
        "@type": "Question",
        name: faq.question,
        acceptedAnswer: { "@type": "Answer", text: faq.answer },
      })),
    });
  }
  return result;
}

async function renderPage(config: SiteConfig, sitePage: SitePage, options: RenderOptions): Promise<RenderedPage> {
  // Serialized for hydration too, so server and client agree on every link.
  const site = sitePage.data.site ? { ...sitePage.data.site, turnstileFrame: options.turnstileFrame ?? null } : sitePage.data.site;
  const page = {
    ...sitePage,
    data: { ...sitePage.data, site, linkPrefix: options.linkPrefix, preview: options.preview },
  };
  const stream = await renderToReadableStream(<SiteView data={page.data} />);
  await stream.allReady;
  const body = await new Response(stream).text();
  const hydrate = options.hydrate ?? true;
  const { title, description } = config.meta;
  const lang = ("lang" in config.meta && config.meta.lang) || "ko";
  const home = page.route === "/";
  const pageTitle = home ? title : `${page.title} | ${isServiceConfig(config) ? config.business.name : title}`;
  const url = options.origin ? `${options.origin}${page.route === "/" ? "/" : `${page.route}`}` : undefined;
  const hero = page.data.page.blocks[0];
  const heroImage = absolute(
    options.origin,
    hero?.__component === "blocks.hero" ? (hero.image?.media as { url?: string } | undefined)?.url : undefined,
  );
  const head = [
    `<title>${escapeHtml(pageTitle)}</title>`,
    `<meta name="description" content="${escapeHtml(description)}">`,
    options.preview ? `<meta name="robots" content="noindex, nofollow">` : "",
    url && !options.preview ? `<link rel="canonical" href="${escapeHtml(url)}">` : "",
    `<meta property="og:type" content="website">`,
    `<meta property="og:title" content="${escapeHtml(pageTitle)}">`,
    `<meta property="og:description" content="${escapeHtml(description)}">`,
    url ? `<meta property="og:url" content="${escapeHtml(url)}">` : "",
    heroImage ? `<meta property="og:image" content="${escapeHtml(heroImage)}">` : "",
    `<meta property="og:locale" content="ko_KR">`,
    ...(home ? structuredData(config, options.origin, heroImage) : []).map(
      (data) => `<script type="application/ld+json">${escapeJson(data)}</script>`,
    ),
  ]
    .filter(Boolean)
    .join("\n");
  const html = `<!doctype html>
<html lang="${lang}">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
${head}
${lcpPreload(body)}
<link rel="stylesheet" href="${options.assetBase}/_assets/site.css">
</head>
<body${page.data.site ? ' class="has-contact-bar"' : ""}>
<div id="root">${body}</div>
${hydrate ? `<script id="${DATA_ID}" type="application/json">${escapeJson(page.data)}</script>
<script type="module" src="${options.assetBase}/_assets/client.js"></script>` : ""}
</body>
</html>`;
  return { route: page.route, path: page.route === "/" ? "index.html" : `${page.route.slice(1)}/index.html`, html };
}

/** Config → one HTML document per page. Runs in workerd and Node. */
export async function renderSite(input: SiteConfigInput, options: RenderOptions): Promise<RenderResult> {
  const started = Date.now();
  const config = requireValidConfig(input);
  const pages: RenderedPage[] = [];
  for (const page of sitePagesFor(config, options.assetBase)) pages.push(await renderPage(config, page, options));
  return { html: pages[0]!.html, pages, ms: Date.now() - started };
}

  );

    `<title>${escapeHtml(pageTitle)}</title>`,
    `<meta name="description" content="${escapeHtml(description)}">`,
    options.preview ? `<meta name="robots" content="noindex, nofollow">` : "",
    url && !options.preview ? `<link rel="canonical" href="${escapeHtml(url)}">` : "",
    `<meta property="og:type" content="website">`,
    `<meta property="og:title" content="${escapeHtml(pageTitle)}">`,
    `<meta property="og:description" content="${escapeHtml(description)}">`,
    url ? `<meta property="og:url" content="${escapeHtml(url)}">` : "",
    heroImage ? `<meta property="og:image" content="${escapeHtml(heroImage)}">` : "",
    `<meta property="og:locale" content="ko_KR">`,
    ...(home ? structuredData(config, options.origin, heroImage) : []).map(
      (data) => `<script type="application/ld+json">${escapeJson(data)}</script>`,
    ),
  ]
    .filter(Boolean)
    .join("\n");
  const html = `<!doctype html>
<html lang="${lang}">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
${head}
${lcpPreload(body)}
<link rel="stylesheet" href="${options.assetBase}/_assets/site.css">
</head>
<body${page.data.site ? ' class="has-contact-bar"' : ""}>
<div id="root">${body}</div>
${hydrate ? `<script id="${DATA_ID}" type="application/json">${escapeJson(page.data)}</script>
<script type="module" src="${options.assetBase}/_assets/client.js"></script>` : ""}
</body>
</html>`;
  return { route: page.route, path: page.route === "/" ? "index.html" : `${page.route.slice(1)}/index.html`, html };
}

/** Config → one HTML document per page. Runs in workerd and Node. */
export async function renderSite(input: SiteConfigInput, options: RenderOptions): Promise<RenderResult> {
  const started = Date.now();
  const config = requireValidConfig(input);
  const pages: RenderedPage[] = [];
  for (const page of sitePagesFor(config, options.assetBase)) pages.push(await renderPage(config, page, options));
  return { html: pages[0]!.html, pages, ms: Date.now() - started };
}
