## Anti-Patterns to Flag

| Anti-pattern | Consequence and preferred pattern |
|-------------|-----------------------------------|
| `await response.text()` or similar buffering on unbounded data | Can exhaust Worker memory; [stream large or unbounded bodies](references/runtime-patterns.md#stream-request-and-response-bodies). |
| Hardcoded secrets in source or config | Leaks credentials through version control; use Wrangler secrets. |
| `Math.random()` for security-sensitive tokens or IDs | Predictable values; use `crypto.randomUUID()` or `crypto.getRandomValues()`. |
| Async work started without awaiting, returning, or attaching it to `ctx.waitUntil()` | Work can be dropped and errors missed; tie it to the request or background-work lifetime. |
| Module-level mutable request state | Leaks data across requests and can cause I/O ownership errors; pass request state explicitly. |
| Cloudflare REST API calls for operations available through Worker bindings | Adds network and authentication overhead; use the available binding. |
| `ctx.passThroughOnException()` used as general error handling | Can conceal Worker failures by forwarding to the origin; use explicit error handling and structured error responses. |
| Hand-written `Env` that duplicates Wrangler bindings | Can drift from configuration; generate binding types with `wrangler types`. |
| Direct string comparison of secret values | Can expose timing differences; use the [Web Crypto comparison pattern](references/runtime-patterns.md#use-web-crypto-for-secure-token-generation). |
| Destructuring `ctx` methods, such as `const { waitUntil } = ctx` | Loses the receiver; call `ctx.waitUntil(...)`. |
| `any` on `Env` or handler parameters | Hides binding and handler contract errors; use the project's generated and platform types. |
| `as unknown as T` to force a platform type match | Hides incompatibilities; fix the underlying contract. |
| `implements` used in place of extending a platform base class | Does not inherit runtime behavior, `this.ctx`, or `this.env`; use the appropriate base class. |
| Unbound `env.X` in a platform class method | Bindings are available through `this.env.X`; see [binding access patterns](references/platform-apis.md#binding-access--the-most-common-error). |
| Applying one serialization rule across Queues, Workflow steps, storage, and WebSockets | Can reject valid payloads or accept unsupported ones; check the [specific API and encoding](references/platform-apis.md#serialization-boundaries). |

/** Keeps the feed row in step with the live version: present with its blog URL, or removed. */
export async function syncBlogFeed(env: Env, siteId: string, config: { business?: { blogUrl?: string } }) {
  const blogUrl = config.business?.blogUrl ?? null;
  if (!blogUrl) {
    await env.DB.prepare("DELETE FROM blog_feeds WHERE site_id = ?").bind(siteId).run();
    await env.ROUTES.delete(kvBlogKey(siteId));
    return;
  }
  // A new or changed URL is due immediately (fetched_at NULL).
  await env.DB.prepare(
    `INSERT INTO blog_feeds (site_id, blog_url) VALUES (?, ?)
     ON CONFLICT (site_id) DO UPDATE SET blog_url = excluded.blog_url,
       fetched_at = CASE WHEN blog_feeds.blog_url = excluded.blog_url THEN blog_feeds.fetched_at ELSE NULL END`,
  )
    .bind(siteId, blogUrl)
    .run();
}

/** Fetches one feed and caches its latest posts. Failures keep the previous cache. */
export async function refreshBlog(env: Env, siteId: string) {
  const row = await env.DB.prepare("SELECT blog_url FROM blog_feeds WHERE site_id = ?").bind(siteId).first<{ blog_url: string }>();
  if (!row) throw new HttpError(404, "블로그가 연결된 게시 버전이 없습니다");
  const rss = naverRssUrl(row.blog_url, env.BLOG_RSS_BASE);
  const now = new Date().toISOString();
  try {
    if (!rss) throw new Error("네이버 블로그 주소가 아닙니다");
    const res = await fetch(rss, { headers: { "user-agent": "dsites-blog-feed/1.0", accept: "application/rss+xml, text/xml" } });
    if (!res.ok) throw new Error(`RSS ${res.status}`);
    const xml = new TextDecoder().decode(await readLimited(res.body, 2 * 1024 * 1024));
    const feed: BlogFeed = { fetchedAt: now, posts: parseNaverRss(xml) };
    await env.ROUTES.put(kvBlogKey(siteId), JSON.stringify(feed));
    await env.DB.prepare("UPDATE blog_feeds SET fetched_at = ?, status = 'ok', error = NULL, post_count = ? WHERE site_id = ?")
      .bind(now, feed.posts.length, siteId)
      .run();
    return { status: "ok", posts: feed.posts.length };
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    await env.DB.prepare("UPDATE blog_feeds SET fetched_at = ?, status = 'error', error = ? WHERE site_id = ?")
      .bind(now, message.slice(0, 300), siteId)
      .run();
    return { status: "error", error: message };
  }
}

/** Cron: refresh up to 20 feeds that are new or older than three hours. */
export async function refreshDueBlogs(env: Env) {
  const { results } = await env.DB.prepare(
    "SELECT site_id FROM blog_feeds WHERE fetched_at IS NULL OR fetched_at < ? ORDER BY fetched_at LIMIT 20",
  )
    .bind(new Date(Date.now() - STALE_MS).toISOString())
    .all<{ site_id: string }>();
  let ok = 0;
  for (const { site_id } of results) if ((await refreshBlog(env, site_id)).status === "ok") ok++;
  return { due: results.length, ok };
}

/** `https://blog.naver.com/<id>` → `<base>/<id>.xml` (base: `https://rss.blog.naver.com`). */
export function naverRssUrl(blogUrl: string, base = "https://rss.blog.naver.com"): string | null {
  const id = blogUrl.match(/^https:\/\/(?:m\.)?blog\.naver\.com\/([A-Za-z0-9_-]+)\/?$/)?.[1];
  return id ? `${base}/${id}.xml` : null;
}

const ENTITIES: Record<string, string> = { amp: "&", lt: "<", gt: ">", quot: '"', apos: "'", nbsp: " " };

function decode(value: string) {
  return value
    .replace(/<!\[CDATA\[([\s\S]*?)\]\]>/g, "$1")
    .replace(/&(#x[0-9a-f]+|#\d+|[a-z]+);/gi, (whole, code: string) => {
      if (code[0] === "#") {
        const n = code[1] === "x" || code[1] === "X" ? parseInt(code.slice(2), 16) : parseInt(code.slice(1), 10);
        return Number.isFinite(n) ? String.fromCodePoint(n) : whole;
      }
      return ENTITIES[code.toLowerCase()] ?? whole;
    });
}

const text = (html: string) => html.replace(/<[^>]+>/g, " ").replace(/\s+/g, " ").trim();

/** Newest posts from a Naver blog RSS document. Only https links and pstatic images survive. */
export function parseNaverRss(xml: string, limit = BLOG_POSTS): BlogPost[] {
  const items = xml.match(/<item>[\s\S]*?<\/item>/g) ?? [];
  const posts: BlogPost[] = [];
  for (const item of items) {
    const title = text(decode(tag(item, "title")));
    const link = decode(tag(item, "link")).trim();
    if (!title || !/^https:\/\/(m\.)?blog\.naver\.com\//.test(link)) continue;
    const description = decode(tag(item, "description"));
    const image = description.match(/https:\/\/(?:blogthumb|postfiles)\.pstatic\.net\/[^"'\s<>]+/)?.[0] ?? null;
    const pub = Date.parse(decode(tag(item, "pubDate")));
    const summary = text(description);
    posts.push({
      title: title.slice(0, 120),
      link: link.replace(/\?.*$/, ""),
      date: Number.isFinite(pub) ? new Date(pub).toISOString() : null,
      summary: summary.length > 110 ? `${summary.slice(0, 110)}…` : summary,
      image,
    });
    if (posts.length >= limit) break;
  }
  return posts;
}

 * Renders every page of a version to R2 unless it is already rendered against the current
 * shared assets and origin.

/** Everything outside the config that changes the HTML (stored in rendered_origin). */
export const renderKeyOf = (env: Env, origin: string) => JSON.stringify({ origin, turnstileFrame: turnstileFrameUrl(env) });

async function ensureRendered(env: Env, version: VersionRow, origin: string, { keepAssets = false } = {}) {
  const config = JSON.parse(version.config) as SiteConfig;
  const turnstileFrame = turnstileFrameUrl(env);
  const renderKey = renderKeyOf(env, origin);
  // Rollback keeps the version's own shared assets; publish moves it to the current ones.
  const assetsOk = keepAssets || version.asset_version === ASSET_VERSION;
  if (version.rendered_at && assetsOk && version.rendered_origin === renderKey) {
    return routesOf(config);
  }
  // HTML comes from the current renderer, so it must point at the current client bundle.
  await requireSharedAssets(env);
  const { pages } = await renderSite(config, { assetBase: assetBase(), origin, turnstileFrame });
  // A fresh, immutable prefix per render: concurrent renders never mix pages, and the pointer
  // switches to a complete set only after markRendered.
  const renderId = crypto.randomUUID().replace(/-/g, "").slice(0, 16);
  for (const page of pages) {
    await env.SITES.put(r2PageKey(version.site_id, version.id, page.path, renderId), page.html, {
      httpMetadata: { contentType: "text/html; charset=utf-8" },
    });
  }
  await markRendered(env.DB, version.id, ASSET_VERSION, renderKey, renderId);
  return pages.map((page) => page.route);
}

async function pointTo(env: Env, site: SiteRow, version: VersionRow, pages: string[]) {
  const publishedAt = await markPublished(env.DB, site.id, version.id, ASSET_VERSION);
  const renderId = (await getVersion(env.DB, site.id, version.id)).render_id;
  const pointer: PublishedPointer = { versionId: version.id, number: version.number, publishedAt, pages, renderId };
  await env.ROUTES.put(kvSiteKey(site.id), JSON.stringify(pointer));
  await syncHostRecordsNow(env, site.id);
  await syncBlogFeed(env, site.id, JSON.parse(version.config) as { business?: { blogUrl?: string } });
  return pointer;
}

/** Publish = render (if needed) → R2 → move the KV pointer. Visible worldwide within about a minute. */
export async function publish(env: Env, site: SiteRow, versionId?: string) {
  await requireSharedAssets(env);
  const version = versionId ? await getVersion(env.DB, site.id, versionId) : await latestVersion(env.DB, site.id);
  const pages = await ensureRendered(env, version, await siteOrigin(env, site));
  return pointTo(env, site, version, pages);
}

/**
 * Rollback = move the pointer back to a version that was published before. No re-render, unless
 * the site's origin changed since that version was rendered (custom domain added or removed):
 * then only its HTML is re-rendered so canonical and OG URLs point at the current host.
 */
export async function rollback(env: Env, site: SiteRow, versionId: string) {
  const version = await getVersion(env.DB, site.id, versionId);
  if (!version.first_published_at) throw new HttpError(409, "게시된 적 없는 버전으로는 롤백할 수 없습니다");
  const exists = await env.SITES.head(r2PageKey(site.id, version.id, "index.html", version.render_id));
  if (!exists) throw new HttpError(409, "렌더된 파일이 R2에 없습니다. publish로 다시 게시하세요");
  const pages = await ensureRendered(env, version, await siteOrigin(env, site), { keepAssets: true });
  return pointTo(env, site, version, pages);
}

/**
 * The site's origin changed (custom domain activated, failed or removed): re-render the live
 * version's HTML in place (same R2 keys). The KV pointer is not touched, so this can never undo
 * a concurrent publish; if the live version changes meanwhile, the new one is checked too.
 * Returns false when it could not settle (the caller keeps origin_pending for the cron).
 */
export async function rerenderLive(env: Env, siteId: string): Promise<boolean> {
  for (let attempt = 0; attempt < 3; attempt++) {
    const site = await getSite(env.DB, siteId);
    if (!site.published_version_id) {
      await syncHostRecordsNow(env, siteId);
      return true;
    }
    const version = await getVersion(env.DB, siteId, site.published_version_id);
    await ensureRendered(env, version, await siteOrigin(env, site));
    // Settled only if the live version is unchanged and its HTML matches the origin as of now
    // (an overlapping domain change may have moved it while this render ran).
    const after = await getSite(env.DB, siteId);
    const rendered = await getVersion(env.DB, siteId, version.id);
    if (after.published_version_id === version.id && rendered.rendered_origin === renderKeyOf(env, await siteOrigin(env, after))) {
      await syncHostRecordsNow(env, siteId);
      return true;
    }
  }
  return false;
}

/**
 * Preload for the LCP image, copied from the rendered `<img fetchpriority="high">` itself so
 * src, srcset and sizes always match what the image will pick (no double download).
 */
export function lcpPreload(body: string): string {
  const img = body.match(/<img\b[^>]*fetchPriority="high"[^>]*>|<img\b[^>]*fetchpriority="high"[^>]*>/i)?.[0];
  if (!img) return "";
  const attr = (name: string) => img.match(new RegExp(`\\s${name}="([^"]*)"`, "i"))?.[1];
  const src = attr("src");
  if (!src) return "";
  const srcset = attr("srcSet") ?? attr("srcset");
  const sizes = attr("sizes");
  return `<link rel="preload" as="image" href="${src}"${srcset ? ` imagesrcset="${srcset}"` : ""}${sizes ? ` imagesizes="${sizes}"` : ""} fetchpriority="high">`;
}

const MAX_FORM_BYTES = 16 * 1024;

const jsonResponse = (status: number, body: unknown) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { "content-type": "application/json; charset=utf-8", "cache-control": "no-store" },
  });

/** Reads a small JSON body, refusing anything larger than the limit while streaming. */
async function readSmallJson(request: Request, limit: number): Promise<unknown> {
  const reader = request.body?.getReader();
  if (!reader) return null;
  const chunks: Uint8Array[] = [];
  let total = 0;
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    total += value.byteLength;
    if (total > limit) {
      await reader.cancel();
      throw new Error("too large");
    }
    chunks.push(value);
  }
  const bytes = new Uint8Array(total);
  let offset = 0;
  for (const chunk of chunks) {
    bytes.set(chunk, offset);
    offset += chunk.byteLength;
  }
  return JSON.parse(new TextDecoder().decode(bytes));
}

// KV values are cached at the edge for this long (KV minimum); publish/rollback propagate within it.
const KV_CACHE_TTL = 30;

function page(status: number, message: string, headers: Record<string, string> = {}) {
  return new Response(
    `<!doctype html><html lang="ko"><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1"><meta name="robots" content="noindex"><title>${message}</title><body style="font-family:system-ui;padding:48px"><p>${message}</p></body></html>`,
    { status, headers: { "content-type": "text/html; charset=utf-8", "cache-control": "no-store", ...headers } },
  );
}

async function serveObject(env: Env, key: string, cacheControl: string) {
  const object = await env.SITES.get(key);
  if (!object) return null;
  const headers = new Headers();
  object.writeHttpMetadata(headers);
  headers.set("etag", object.httpEtag);
  headers.set("cache-control", cacheControl);
  return new Response(object.body, { headers });
}

/** One KV read for M2 records; M1 records (bare siteId) need a second read for the pointer. */
async function resolveHost(env: Env, hostname: string): Promise<HostRecord | null> {
  const raw = await env.ROUTES.get(kvHostKey(hostname), { cacheTtl: KV_CACHE_TTL });
  if (!raw) return null;
  if (raw.startsWith("{")) return JSON.parse(raw) as HostRecord;
  const pointer = await env.ROUTES.get<PublishedPointer>(kvSiteKey(raw), { type: "json", cacheTtl: KV_CACHE_TTL });
  return { siteId: raw, status: "active", pointer, redirectTo: null };
}

export default {
  async fetch(request: Request, env: Env, ctx: ExecutionContext): Promise<Response> {
    const url = new URL(request.url);
    const hostname = url.hostname.toLowerCase();

    // Pageview and click beacons. Always 204: callers learn nothing, pages never wait on it.
    if (url.pathname === "/api/track") {
      if (request.method !== "POST") return new Response(null, { status: 405 });
      const host = await resolveHost(env, hostname);
      if (host && host.status === "active" && host.pointer) {
        let body: unknown = null;
        try {
          body = await readSmallJson(request, 1024);
        } catch {
          body = null;
        }
        if (body) {
          ctx.waitUntil(env.API.track({ siteId: host.siteId, userAgent: request.headers.get("user-agent"), body }));
        }
      }
      return new Response(null, { status: 204, headers: { "cache-control": "no-store" } });
    }

    // Visitor inquiry: the site is identified by Host here, then handed to the api by RPC.
    if (url.pathname === "/api/inquiry") {
      if (request.method !== "POST") return jsonResponse(405, { ok: false, error: "method not allowed" });
      const host = await resolveHost(env, hostname);
      if (!host || host.status !== "active" || !host.pointer) {
        return jsonResponse(404, { ok: false, error: "문의를 받을 수 없는 사이트입니다" });
      }
      if (!(request.headers.get("content-type") ?? "").startsWith("application/json")) {
        return jsonResponse(415, { ok: false, error: "JSON으로 보내 주세요" });
      }
      let body: unknown;
      try {
        body = await readSmallJson(request, MAX_FORM_BYTES);
      } catch {
        return jsonResponse(400, { ok: false, error: "요청을 읽지 못했습니다" });
      }
      const result = await env.API.submitInquiry({
        siteId: host.siteId,
        hostname,
        ip: request.headers.get("cf-connecting-ip"),
        body,
      });
      return jsonResponse(result.status, result.body);
    }

    if (request.method !== "GET" && request.method !== "HEAD") {
      return new Response("method not allowed", { status: 405 });
    }

    // Shared assets are content-addressed by version, so any host may serve them.
    if (url.pathname.startsWith("/_shared/")) {
      return (
        (await serveObject(env, r2SharedKey(url.pathname.slice("/_shared/".length)), "public, max-age=31536000, immutable")) ??
        new Response("not found", { status: 404 })
      );
    }
    // Uploaded photos are public and content-addressed.
    const media = MEDIA_PATH.exec(url.pathname);
    if (media) {
      return (
        (await serveObject(env, r2MediaKey(media[1]!, media[2]!), "public, max-age=31536000, immutable")) ??
        new Response("not found", { status: 404 })
      );
    }

    const host = await resolveHost(env, hostname);
    if (!host) return page(404, "연결된 사이트가 없습니다.");
    if (host.status === "cancelled") return page(410, "운영이 종료된 사이트입니다.");
    if (host.status === "suspended") return page(503, "잠시 운영을 멈춘 사이트입니다.", { "retry-after": "86400" });
    if (host.redirectTo) {
      return Response.redirect(`${url.protocol}//${host.redirectTo}${url.pathname}${url.search}`, 301);
    }
    const pointer = host.pointer;
    if (!pointer) return page(404, "아직 공개되지 않은 사이트입니다.");

    if (url.pathname === "/robots.txt") {
      return new Response(`User-agent: *\nAllow: /\nSitemap: https://${hostname}/sitemap.xml\n`, {
        headers: { "content-type": "text/plain; charset=utf-8", "cache-control": "public, max-age=3600" },
      });
    }
    const pages = pointer.pages ?? ["/"];
    if (url.pathname === "/sitemap.xml") {
      const lastmod = pointer.publishedAt.slice(0, 10);
      const urls = pages
        .map((route) => `<url><loc>https://${hostname}${route}</loc><lastmod>${lastmod}</lastmod></url>`)
        .join("");
      return new Response(
        `<?xml version="1.0" encoding="UTF-8"?>\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">${urls}</urlset>\n`,
        { headers: { "content-type": "application/xml; charset=utf-8", "cache-control": "public, max-age=3600" } },
      );
    }
    const route = url.pathname === "/index.html" ? "/" : url.pathname.replace(/\/(index\.html)?$/, "") || "/";
    if (pages.includes(route)) {
      const path = route === "/" ? "index.html" : `${route.slice(1)}/index.html`;
      const response = await serveObject(env, r2PageKey(host.siteId, pointer.versionId, path, pointer.renderId), "public, max-age=60");
      if (response) {
        response.headers.set("x-site-version", String(pointer.number));
        if (route !== "/") return response;
        // Latest blog posts, filled at request time from the cron's KV cache. The body now
        // changes without the object changing, so the R2 ETag no longer describes it.
        response.headers.delete("etag");
        const siteId = host.siteId;
        return new HTMLRewriter()
          .on("[data-blog-feed]", {
            async element(element) {
              const feed = await env.ROUTES.get<BlogFeed>(`blog:${siteId}`, { type: "json", cacheTtl: 300 });
              const html = feed ? blogCardsHtml(feed.posts) : "";
              if (html) element.setInnerContent(html, { html: true });
            },
          })
          .transform(response);
      }
      return page(404, "페이지 파일을 찾지 못했습니다.");
    }
    return page(404, "없는 주소입니다.");
  },
};
