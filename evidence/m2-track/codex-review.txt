## Anti-Patterns to Flag

| Anti-pattern | Consequence and preferred pattern |
|-------------|-----------------------------------|
| `await response.text()` or similar buffering on unbounded data | Can exhaust Worker memory; [stream large or unbounded bodies](references/runtime-patterns.md#stream-request-and-response-bodies). |
| Hardcoded secrets in source or config | Leaks credentials through version control; use Wrangler secrets. |
| `Math.random()` for security-sensitive tokens or IDs | Predictable values; use `crypto.randomUUID()` or `crypto.getRandomValues()`. |
| Async work started without awaiting, returning, or attaching it to `ctx.waitUntil()` | Work can be dropped and errors missed; tie it to the request or background-work lifetime. |
| Module-level mutable request state | Leaks data across requests and can cause I/O ownership errors; pass request state explicitly. |
| Cloudflare REST API calls for operations available through Worker bindings | Adds network and authentication overhead; use the available binding. |
| `ctx.passThroughOnException()` used as general error handling | Can conceal Worker failures by forwarding to the origin; use explicit error handling and structured error responses. |
| Hand-written `Env` that duplicates Wrangler bindings | Can drift from configuration; generate binding types with `wrangler types`. |
| Direct string comparison of secret values | Can expose timing differences; use the [Web Crypto comparison pattern](references/runtime-patterns.md#use-web-crypto-for-secure-token-generation). |
| Destructuring `ctx` methods, such as `const { waitUntil } = ctx` | Loses the receiver; call `ctx.waitUntil(...)`. |
| `any` on `Env` or handler parameters | Hides binding and handler contract errors; use the project's generated and platform types. |
| `as unknown as T` to force a platform type match | Hides incompatibilities; fix the underlying contract. |
| `implements` used in place of extending a platform base class | Does not inherit runtime behavior, `this.ctx`, or `this.env`; use the appropriate base class. |
| Unbound `env.X` in a platform class method | Bindings are available through `this.env.X`; see [binding access patterns](references/platform-apis.md#binding-access--the-most-common-error). |
| Applying one serialization rule across Queues, Workflow steps, storage, and WebSockets | Can reject valid payloads or accept unsupported ones; check the [specific API and encoding](references/platform-apis.md#serialization-boundaries). |

/** Click and pageview events the platform measures (stored in D1 `metrics_daily`). */
export type TrackEvent = "pageview" | "tel" | "kakao" | "map" | "booking" | "inquiry_cta";

export const TRACK_ENDPOINT = "/api/track";

/** Classifies a clicked link; `data-track` wins, otherwise the destination decides. */
export function classifyLink(anchor: { getAttribute(name: string): string | null }): TrackEvent | null {
  const explicit = anchor.getAttribute("data-track");
  if (explicit) return explicit as TrackEvent;
  const href = anchor.getAttribute("href") ?? "";
  if (href.startsWith("tel:")) return "tel";
  if (/^https:\/\/(pf|open)\.kakao\.com\//.test(href)) return "kakao";
  if (/^https:\/\/(m\.)?booking\.naver\.com\//.test(href)) return "booking";
  if (/^https:\/\/((m\.)?map\.naver\.com|naver\.me|map\.kakao\.com|kko\.to|maps\.app\.goo\.gl|(www\.)?google\.[a-z.]+\/maps)/.test(href)) {
    return "map";
  }
  if (/(^|\/)#inquiry$/.test(href)) return "inquiry_cta";
  return null;
}

function send(event: TrackEvent) {
  const body = JSON.stringify({ event, page: location.pathname });
  // sendBeacon survives navigation (tel:, external links); text/plain avoids a preflight.
  if (!navigator.sendBeacon?.(TRACK_ENDPOINT, new Blob([body], { type: "text/plain" }))) {
    void fetch(TRACK_ENDPOINT, { method: "POST", body, keepalive: true }).catch(() => undefined);
  }
}

/** Pageview now, then one delegated listener for every tracked link on the page. */
export function startTracking() {
  send("pageview");
  document.addEventListener(
    "click",
    (event) => {
      const anchor = (event.target as Element | null)?.closest?.("a[href]");
      const kind = anchor ? classifyLink(anchor) : null;
      if (kind) send(kind);
    },
    { capture: true },
  );
}

async function serveImmutable(env: Env, key: string) {
  const object = await env.SITES.get(key);
  if (!object) return new Response("not found", { status: 404 });
  const headers = new Headers();
  object.writeHttpMetadata(headers);
  headers.set("cache-control", "public, max-age=31536000, immutable");
  return new Response(object.body, { headers });
}

async function handle(request: Request, env: Env): Promise<Response> {
  const url = new URL(request.url);
  const parts = url.pathname.split("/").filter(Boolean);
  const method = request.method;

  if (method === "GET" && url.pathname === "/health") return json({ ok: true });

  // Shared assets are public and immutable; previews load them from here.
  if (method === "GET" && parts[0] === "_shared") return serveImmutable(env, r2SharedKey(parts.slice(1).join("/")));
  // Uploaded photos (public, content-addressed) so previews show them before publishing.
  const media = method === "GET" ? MEDIA_PATH.exec(url.pathname) : null;
  if (media) return serveImmutable(env, r2MediaKey(media[1]!, media[2]!));

  // Turnstile widget frame embedded by public sites (see turnstile.ts).
  if (method === "GET" && url.pathname === "/turnstile-frame") {
    if (!env.TURNSTILE_SITE_KEY) return new Response("turnstile not configured", { status: 404 });
    const parent = url.searchParams.get("parent") ?? "";
    if (!/^https?:\/\/[a-z0-9.-]+(:\d+)?$/i.test(parent)) return new Response("bad parent", { status: 400 });
    return new Response(turnstileFrame(env.TURNSTILE_SITE_KEY, parent), {
      headers: { "content-type": "text/html; charset=utf-8", "cache-control": "no-store", "x-robots-tag": "noindex" },
    });
  }

  // Preview: anyone with the token can view a version. Never indexed. /preview/<token>[/privacy]
  if (method === "GET" && parts[0] === "preview" && (parts.length === 2 || parts.length === 3)) {
    const version = await versionByPreviewToken(env.DB, parts[1]!);
    const site = await getSite(env.DB, version.site_id);
    const { pages } = await renderSite(JSON.parse(version.config) as SiteConfig, {
      assetBase: assetBase(),
      origin: await siteOrigin(env, site),
      preview: true,
      turnstileFrame: null,
      // Site links are root-relative (`/#inquiry`, `/privacy`); keep them inside this preview.
      linkPrefix: `/preview/${parts[1]}`,
    });
    const route = parts[2] ? `/${parts[2]}` : "/";
    const page = pages.find((entry) => entry.route === route);
    if (!page) throw new HttpError(404, "없는 페이지입니다");
    return new Response(page.html, {
      headers: {
        "content-type": "text/html; charset=utf-8",
        "x-robots-tag": "noindex, nofollow",
        "cache-control": "no-store",
      },
    });
  }

  requireAdmin(request, env.ADMIN_TOKEN);

  // PUT /admin/shared/<version>/<path>: upload one shared asset file.
  if (method === "PUT" && parts[0] === "admin" && parts[1] === "shared" && parts.length > 3) {
    const key = r2SharedKey(parts.slice(2).join("/"));
    await env.SITES.put(key, request.body, {
      httpMetadata: { contentType: request.headers.get("content-type") ?? "application/octet-stream" },
    });
    return json({ key }, 201);
  }

  if (parts[0] !== "sites") throw new HttpError(404, "없는 경로입니다");

  // POST /sites {config, createdBy}
  if (method === "POST" && parts.length === 1) {
    const body = await readJson<{ config: unknown; createdBy?: CreatedBy }>(request);
    const config = validated(body.config);
    const hostname = `${config.slug}.${env.BASE_DOMAIN}`;
    const { siteId, versionId } = await createSite(env.DB, config, creator(body.createdBy), hostname);
    await syncHostRecordsNow(env, siteId);
    return json({ siteId, versionId, hostname }, 201);
  }

  // GET /sites: every site with its published version number and primary hostname.
  if (method === "GET" && parts.length === 1) {
    return json({ sites: await listSites(env.DB, env.BASE_DOMAIN) });
  }
  if (parts.length < 2) throw new HttpError(404, "없는 경로입니다");

  const site = await getSite(env.DB, parts[1]!);

  if (method === "GET" && parts.length === 2) {
    return json({ site, versions: await listVersions(env.DB, site.id) });
  }

  // PATCH /sites/:id {notifyEmail}: operational settings outside the versioned config.
  if (method === "PATCH" && parts.length === 2) {
    const body = await readJson<{ notifyEmail?: unknown }>(request);
    if ("notifyEmail" in body) {
      await env.DB.prepare("UPDATE sites SET notify_email = ? WHERE id = ?").bind(requireEmail(body.notifyEmail), site.id).run();
    }
    return json({ site: await getSite(env.DB, site.id) });
  }

  // GET /sites/:id/metrics?from=YYYY-MM-DD&to=YYYY-MM-DD (Korean dates, inclusive; default last 30 days)
  if (method === "GET" && parts[2] === "metrics" && parts.length === 3) {
    const today = kstDay(new Date());
    const from = url.searchParams.get("from") ?? kstDay(new Date(Date.now() - 29 * 86_400_000));
    const to = url.searchParams.get("to") ?? today;
    if (![from, to].every((d) => /^\d{4}-\d{2}-\d{2}$/.test(d))) throw new HttpError(400, "from·to는 YYYY-MM-DD");
    return json(await metricsReport(env.DB, site.id, from, to));
  }

  // GET /sites/:id/inquiries[.csv]?from=&to= (ISO dates)
  if (method === "GET" && (parts[2] === "inquiries" || parts[2] === "inquiries.csv") && parts.length === 3) {
    const range = { from: url.searchParams.get("from") ?? undefined, to: url.searchParams.get("to") ?? undefined };
    if (parts[2] === "inquiries") {
      return json(
        await listInquiries(env, site.id, {
          ...range,
          before: url.searchParams.get("before") ?? undefined,
          limit: Number(url.searchParams.get("limit") ?? 200),
        }),
      );
    }
    return new Response(inquiriesCsv(await allInquiries(env, site.id, range)), {
      headers: {
        "content-type": "text/csv; charset=utf-8",
        "content-disposition": `attachment; filename="inquiries-${site.slug}.csv"`,
      },
    });
  }

  // POST /sites/:id/versions {config, createdBy, note}: new draft version
  if (method === "POST" && parts[2] === "versions") {
    const body = await readJson<{ config: unknown; createdBy?: CreatedBy; note?: string }>(request);
    const result = await addVersion(env.DB, site, validated(body.config), creator(body.createdBy), body.note ?? null);
    return json(result, 201);
  }

  // POST /sites/:id/assets: raw image body, or JSON {url}. GET lists them.
  if (parts[2] === "assets" && parts.length === 3) {
    if (method === "POST") return json(await uploadAsset(env, site, request), 201);
    if (method === "GET") return json({ assets: await listAssets(env, site) });
  }

  // POST /sites/:id/publish {versionId?}: latest version when omitted
  if (method === "POST" && parts[2] === "publish") {
    const body = await readJson<{ versionId?: string }>(request).catch(() => ({}) as { versionId?: string });
    return json(await publish(env, site, body.versionId));
  }

  // POST /sites/:id/suspend | /resume: stop or restart public serving (non-payment, requests).
  if (method === "POST" && (parts[2] === "suspend" || parts[2] === "resume") && parts.length === 3) {
    if (site.cancelled_at) throw new HttpError(409, "해지된 사이트입니다");
    const status = parts[2] === "suspend" ? "suspended" : "active";
    await setSiteStatus(env.DB, site.id, status);
    await syncHostRecordsNow(env, site.id);
    return json({ siteId: site.id, status });
  }

  // POST /sites/:id/rollback {versionId}
  if (method === "POST" && parts[2] === "rollback") {
    const body = await readJson<{ versionId: string }>(request);
    if (!body.versionId) throw new HttpError(400, "versionId가 필요합니다");
    return json(await rollback(env, site, body.versionId));
  }

  throw new HttpError(404, "없는 경로입니다");
}

/**
 * Turnstile token from the platform-hosted frame. The frame loads only after the visitor starts
 * filling the form, so the challenge script never weighs on page load.
 */
function useTurnstile(frameUrl: string | null, active: boolean) {
  const [token, setToken] = useState<string | null>(null);
  const [failed, setFailed] = useState(false);
  const [round, setRound] = useState(0);
  useEffect(() => {
    if (!frameUrl || !active) return;
    const frameOrigin = new URL(frameUrl, location.href).origin;
    function onMessage(event: MessageEvent) {
      if (event.origin !== frameOrigin || event.data?.type !== "dsites-turnstile") return;
      if (event.data.error) setFailed(true);
      else setToken(event.data.token ?? null);
    }
    window.addEventListener("message", onMessage);
    return () => window.removeEventListener("message", onMessage);
  }, [frameUrl, active]);
  const src = frameUrl && active ? `${frameUrl}?parent=${encodeURIComponent(location.origin)}&round=${round}` : null;
  // Tokens are single-use: reloading the frame issues a fresh one.
  const reset = () => {
    setToken(null);
    setFailed(false);
    setRound((n) => n + 1);
  };
  return { token, failed, src, required: Boolean(frameUrl), reset };
}

/** Reads a small JSON body, refusing anything larger than the limit while streaming. */
async function readSmallJson(request: Request, limit: number): Promise<unknown> {
  const reader = request.body?.getReader();
  if (!reader) return null;
  const chunks: Uint8Array[] = [];
  let total = 0;
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    total += value.byteLength;
    if (total > limit) {
      await reader.cancel();
      throw new Error("too large");
    }
    chunks.push(value);
  }
  const bytes = new Uint8Array(total);
  let offset = 0;
  for (const chunk of chunks) {
    bytes.set(chunk, offset);
    offset += chunk.byteLength;
  }
  return JSON.parse(new TextDecoder().decode(bytes));
}

/** One KV read for M2 records; M1 records (bare siteId) need a second read for the pointer. */
async function resolveHost(env: Env, hostname: string): Promise<HostRecord | null> {
  const raw = await env.ROUTES.get(kvHostKey(hostname), { cacheTtl: KV_CACHE_TTL });
  if (!raw) return null;
  if (raw.startsWith("{")) return JSON.parse(raw) as HostRecord;
  const pointer = await env.ROUTES.get<PublishedPointer>(kvSiteKey(raw), { type: "json", cacheTtl: KV_CACHE_TTL });
  return { siteId: raw, status: "active", pointer, redirectTo: null };
}

export default {
  async fetch(request: Request, env: Env, ctx: ExecutionContext): Promise<Response> {
    const url = new URL(request.url);
    const hostname = url.hostname.toLowerCase();

    // Pageview and click beacons. Always 204: callers learn nothing, pages never wait on it.
    if (url.pathname === "/api/track") {
      if (request.method !== "POST") return new Response(null, { status: 405 });
      const host = await resolveHost(env, hostname);
      if (host && host.status === "active" && host.pointer) {
        let body: unknown = null;
        try {
          body = await readSmallJson(request, 1024);
        } catch {
          body = null;
        }
        if (body) {
          ctx.waitUntil(env.API.track({ siteId: host.siteId, userAgent: request.headers.get("user-agent"), body }));
        }
      }
      return new Response(null, { status: 204, headers: { "cache-control": "no-store" } });
    }

    // Visitor inquiry: the site is identified by Host here, then handed to the api by RPC.
    if (url.pathname === "/api/inquiry") {
      if (request.method !== "POST") return jsonResponse(405, { ok: false, error: "method not allowed" });
      const host = await resolveHost(env, hostname);
      if (!host || host.status !== "active" || !host.pointer) {
        return jsonResponse(404, { ok: false, error: "문의를 받을 수 없는 사이트입니다" });
      }
      if (!(request.headers.get("content-type") ?? "").startsWith("application/json")) {
        return jsonResponse(415, { ok: false, error: "JSON으로 보내 주세요" });
      }
      let body: unknown;
      try {
        body = await readSmallJson(request, MAX_FORM_BYTES);
      } catch {
        return jsonResponse(400, { ok: false, error: "요청을 읽지 못했습니다" });
      }
      const result = await env.API.submitInquiry({
        siteId: host.siteId,
        hostname,
        ip: request.headers.get("cf-connecting-ip"),
        body,
      });
      return jsonResponse(result.status, result.body);
    }

    if (request.method !== "GET" && request.method !== "HEAD") {
      return new Response("method not allowed", { status: 405 });
    }

    // Shared assets are content-addressed by version, so any host may serve them.
    if (url.pathname.startsWith("/_shared/")) {
      return (
        (await serveObject(env, r2SharedKey(url.pathname.slice("/_shared/".length)), "public, max-age=31536000, immutable")) ??
        new Response("not found", { status: 404 })
      );
    }
    // Uploaded photos are public and content-addressed.
    const media = MEDIA_PATH.exec(url.pathname);
    if (media) {
      return (
        (await serveObject(env, r2MediaKey(media[1]!, media[2]!), "public, max-age=31536000, immutable")) ??
        new Response("not found", { status: 404 })
      );
    }

    const host = await resolveHost(env, hostname);
    if (!host) return page(404, "연결된 사이트가 없습니다.");
    if (host.status === "cancelled") return page(410, "운영이 종료된 사이트입니다.");
    if (host.status === "suspended") return page(503, "잠시 운영을 멈춘 사이트입니다.", { "retry-after": "86400" });
    if (host.redirectTo) {
      return Response.redirect(`${url.protocol}//${host.redirectTo}${url.pathname}${url.search}`, 301);
    }
    const pointer = host.pointer;
    if (!pointer) return page(404, "아직 공개되지 않은 사이트입니다.");

    if (url.pathname === "/robots.txt") {
      return new Response(`User-agent: *\nAllow: /\nSitemap: https://${hostname}/sitemap.xml\n`, {
        headers: { "content-type": "text/plain; charset=utf-8", "cache-control": "public, max-age=3600" },
      });
    }
    const pages = pointer.pages ?? ["/"];
    if (url.pathname === "/sitemap.xml") {
      const lastmod = pointer.publishedAt.slice(0, 10);
      const urls = pages
        .map((route) => `<url><loc>https://${hostname}${route}</loc><lastmod>${lastmod}</lastmod></url>`)
        .join("");
      return new Response(
        `<?xml version="1.0" encoding="UTF-8"?>\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">${urls}</urlset>\n`,
        { headers: { "content-type": "application/xml; charset=utf-8", "cache-control": "public, max-age=3600" } },
      );
    }
    const route = url.pathname === "/index.html" ? "/" : url.pathname.replace(/\/(index\.html)?$/, "") || "/";
    if (pages.includes(route)) {
      const path = route === "/" ? "index.html" : `${route.slice(1)}/index.html`;
      const response = await serveObject(env, r2PageKey(host.siteId, pointer.versionId, path), "public, max-age=60");
      if (response) {
        response.headers.set("x-site-version", String(pointer.number));
        return response;
      }
      return page(404, "페이지 파일을 찾지 못했습니다.");
    }
    return page(404, "없는 주소입니다.");
  },
};

export const METRIC_EVENTS = ["pageview", "tel", "kakao", "map", "booking", "inquiry_cta", "inquiry"] as const;
export type MetricEvent = (typeof METRIC_EVENTS)[number];

/** Korean calendar day (UTC+9) for daily counters: reports are read in Korean days. */
export const kstDay = (at: Date) => new Date(at.getTime() + 9 * 3600_000).toISOString().slice(0, 10);

export async function bumpMetric(db: D1Database, siteId: string, event: MetricEvent, at = new Date()) {
  await db
    .prepare(
      "INSERT INTO metrics_daily (site_id, date, event, count) VALUES (?, ?, ?, 1) ON CONFLICT (site_id, date, event) DO UPDATE SET count = count + 1",
    )
    .bind(siteId, kstDay(at), event)
    .run();
}

/** Events visitors' browsers may report; `inquiry` is counted by the server on a stored inquiry. */
const CLIENT_EVENTS = new Set<MetricEvent>(["pageview", "tel", "kakao", "map", "booking", "inquiry_cta"]);

/** Crawlers, link previews, monitors and headless browsers do not count as visits. */
export const BOT_UA =
  /bot|crawl|spider|slurp|bingpreview|facebookexternalhit|kakaotalk-scrap|yeti|daumoa|headless|lighthouse|pagespeed|curl|wget|python|node-fetch|axios|go-http|java\//i;

export interface TrackRequest {
  siteId: string;
  userAgent: string | null;
  body: unknown;
}

/** Records one beacon. Returns what happened, for logs and tests; the router always answers 204. */
export async function recordTrack(db: D1Database, request: TrackRequest): Promise<"counted" | "ignored"> {
  const event = (request.body as { event?: unknown } | null)?.event;
  if (typeof event !== "string" || !CLIENT_EVENTS.has(event as MetricEvent)) return "ignored";
  if (!request.userAgent || BOT_UA.test(request.userAgent)) return "ignored";
  const site = await db.prepare("SELECT status, cancelled_at FROM sites WHERE id = ?")
    .bind(request.siteId)
    .first<{ status: string; cancelled_at: string | null }>();
  if (!site || site.status !== "active" || site.cancelled_at) return "ignored";
  await bumpMetric(db, request.siteId, event as MetricEvent);
  return "counted";
}

/** Totals and daily rows for a KST date range (inclusive), for reports and the console. */
export async function metricsReport(db: D1Database, siteId: string, from: string, to: string) {
  const { results } = await db
    .prepare("SELECT date, event, count FROM metrics_daily WHERE site_id = ? AND date >= ? AND date <= ? ORDER BY date, event")
    .bind(siteId, from, to)
    .all<{ date: string; event: MetricEvent; count: number }>();
  const totals = Object.fromEntries(METRIC_EVENTS.map((event) => [event, 0])) as Record<MetricEvent, number>;
  for (const row of results) totals[row.event] += row.count;
  return { from, to, totals, daily: results };
}
