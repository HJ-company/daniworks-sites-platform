/**
 * Full republish after a renderer or template update, in three explicit steps so a visual
 * check can sit between rendering and serving: plan → stage (render, not served) → commit.
 */
export async function republishPlan(env: Env) {
  const { results } = await env.DB.prepare(
    `SELECT s.id, s.slug, s.published_version_id AS version_id, v.number, v.asset_version
     FROM sites s JOIN site_versions v ON v.id = s.published_version_id
     WHERE s.cancelled_at IS NULL ORDER BY s.slug`,
  ).all<{ id: string; slug: string; version_id: string; number: number; asset_version: string | null }>();
  const sites = [];
  for (const row of results) {
    const site = await getSite(env.DB, row.id);
    sites.push({
      siteId: row.id,
      slug: row.slug,
      versionId: row.version_id,
      number: row.number,
      assetVersion: row.asset_version,
      outdated: row.asset_version !== ASSET_VERSION,
      liveUrl: `${env.PUBLIC_SCHEME ?? "https"}://${await primaryHostname(env.DB, site, env.BASE_DOMAIN)}/`,
    });
  }
  return { assetVersion: ASSET_VERSION, total: sites.length, outdated: sites.filter((s) => s.outdated).length, sites };
}

/** Renders the live version with the current renderer under a new render_id. Not served. */
export async function stageRepublish(env: Env, siteId: string) {
  const site = await getSite(env.DB, siteId);
  if (!site.published_version_id) throw new HttpError(409, "게시된 버전이 없습니다");
  const version = await getVersion(env.DB, siteId, site.published_version_id);
  const render = await renderToR2(env, version, await siteOrigin(env, site));
  await env.DB.prepare(
    "INSERT INTO staged_renders (render_id, site_id, version_id, render_key, asset_version, base_render_id, created_at) VALUES (?, ?, ?, ?, ?, ?, ?)",
  )
    .bind(render.renderId, siteId, version.id, render.renderKey, ASSET_VERSION, version.render_id, new Date().toISOString())
    .run();
  return {
    siteId,
    versionId: version.id,
    renderId: render.renderId,
    stagedPath: `/staged/${siteId}/${render.renderId}/`,
    /** Every route in the render; all must look the same before committing. */
    routes: render.routes,
  };
}

/**
 * Serves the staged render (switched in only after a visual check). Valid only while that
 * version is still live and its origin unchanged, so an older render cannot replace newer work.
 */
export async function commitRepublish(env: Env, siteId: string, renderId: string) {
  const staged = await env.DB.prepare("SELECT * FROM staged_renders WHERE render_id = ? AND site_id = ?")
    .bind(renderId, siteId)
    .first<{ version_id: string; render_key: string; asset_version: string; base_render_id: string | null }>();
  if (!staged) throw new HttpError(404, "스테이지 렌더가 없습니다");
  // Staged before a renderer deployment: committing it would downgrade newer renders.
  if (staged.asset_version !== ASSET_VERSION) throw new HttpError(409, "이전 배포에서 만든 스테이지입니다. 다시 스테이지하세요");
  const site = await getSite(env.DB, siteId);
  if (site.published_version_id !== staged.version_id) throw new HttpError(409, "그사이 다른 버전이 게시되었습니다. 다시 스테이지하세요");
  if (staged.render_key !== renderKeyOf(env, await siteOrigin(env, site))) {
    throw new HttpError(409, "그사이 사이트 주소가 바뀌었습니다. 다시 스테이지하세요");
  }
  const now = new Date().toISOString();
  // One transaction: switch the render (fenced on the live render at staging time — a domain
  // change or publish since then wins), and only if that applied, mark the site for route
  // reconciliation and clear its stages. A failure after this still converges via the cron.
  const applied = `EXISTS (SELECT 1 FROM site_versions WHERE id = ? AND render_id = ?)`;
  const [updated] = await env.DB.batch([
    env.DB.prepare(
      `UPDATE site_versions SET render_id = ?, rendered_origin = ?, asset_version = ?, rendered_at = ?
       WHERE id = ? AND render_id IS ? AND id = (SELECT published_version_id FROM sites WHERE id = ?)`,
    ).bind(renderId, staged.render_key, staged.asset_version, now, staged.version_id, staged.base_render_id, siteId),
    env.DB.prepare(
      `UPDATE sites SET template_version = ?, updated_at = ?, routes_synced_at = NULL WHERE id = ? AND ${applied}`,
    ).bind(staged.asset_version, now, siteId, staged.version_id, renderId),
    env.DB.prepare(`DELETE FROM staged_renders WHERE site_id = ? AND ${applied}`).bind(siteId, staged.version_id, renderId),
  ]);
  if (updated?.meta.changes !== 1) throw new HttpError(409, "그사이 게시나 주소 변경이 있었습니다. 다시 스테이지하세요");
  await syncHostRecordsNow(env, siteId);
  return { siteId, versionId: staged.version_id, renderId, assetVersion: staged.asset_version };
}

/** Share of pixels that differ noticeably; a size change counts as fully different. */
async function diff(a, b, name) {
  const [ma, mb] = await Promise.all([a, b].map((png) => sharp(png).metadata()));
  const width = Math.max(ma.width, mb.width);
  const height = Math.max(ma.height, mb.height);
  const sized = ma.width !== mb.width || ma.height !== mb.height;
  // Different sizes (layout change): pad both onto the same canvas so the diff image still shows where.
  const raw = (png, m) =>
    sharp(png)
      .ensureAlpha()
      .extend({ right: width - m.width, bottom: height - m.height, background: { r: 255, g: 0, b: 255, alpha: 1 } })
      .raw()
      .toBuffer({ resolveWithObject: true });
  const [ra, rb] = await Promise.all([raw(a, ma), raw(b, mb)]);
  const mask = Buffer.alloc(width * height * 4);
  let changed = 0;
  for (let i = 0; i < ra.data.length; i += 4) {
    const d = Math.max(Math.abs(ra.data[i] - rb.data[i]), Math.abs(ra.data[i + 1] - rb.data[i + 1]), Math.abs(ra.data[i + 2] - rb.data[i + 2]));
    if (d > 24) {
      changed++;
      mask[i] = 255;
      mask[i + 3] = 255;
    } else {
      mask[i] = mask[i + 1] = mask[i + 2] = ra.data[i] >> 2;
      mask[i + 3] = 255;
    }
  }
  // A size change is never "the same", however few pixels differ.
  const ratio = sized ? Math.max(changed / (width * height), 1) : changed / (width * height);
  if (ratio > threshold) await sharp(mask, { raw: { width, height, channels: 4 } }).png().toFile(`${out}/${name}-diff.png`);
  return sized ? { ratio, reason: `size ${ma.width}x${ma.height} → ${mb.width}x${mb.height}` } : { ratio };
}

## Anti-Patterns to Flag

| Anti-pattern | Consequence and preferred pattern |
|-------------|-----------------------------------|
| `await response.text()` or similar buffering on unbounded data | Can exhaust Worker memory; [stream large or unbounded bodies](references/runtime-patterns.md#stream-request-and-response-bodies). |
| Hardcoded secrets in source or config | Leaks credentials through version control; use Wrangler secrets. |
| `Math.random()` for security-sensitive tokens or IDs | Predictable values; use `crypto.randomUUID()` or `crypto.getRandomValues()`. |
| Async work started without awaiting, returning, or attaching it to `ctx.waitUntil()` | Work can be dropped and errors missed; tie it to the request or background-work lifetime. |
| Module-level mutable request state | Leaks data across requests and can cause I/O ownership errors; pass request state explicitly. |
| Cloudflare REST API calls for operations available through Worker bindings | Adds network and authentication overhead; use the available binding. |
| `ctx.passThroughOnException()` used as general error handling | Can conceal Worker failures by forwarding to the origin; use explicit error handling and structured error responses. |
| Hand-written `Env` that duplicates Wrangler bindings | Can drift from configuration; generate binding types with `wrangler types`. |
| Direct string comparison of secret values | Can expose timing differences; use the [Web Crypto comparison pattern](references/runtime-patterns.md#use-web-crypto-for-secure-token-generation). |
| Destructuring `ctx` methods, such as `const { waitUntil } = ctx` | Loses the receiver; call `ctx.waitUntil(...)`. |
| `any` on `Env` or handler parameters | Hides binding and handler contract errors; use the project's generated and platform types. |
| `as unknown as T` to force a platform type match | Hides incompatibilities; fix the underlying contract. |
| `implements` used in place of extending a platform base class | Does not inherit runtime behavior, `this.ctx`, or `this.env`; use the appropriate base class. |
| Unbound `env.X` in a platform class method | Bindings are available through `this.env.X`; see [binding access patterns](references/platform-apis.md#binding-access--the-most-common-error). |
| Applying one serialization rule across Queues, Workflow steps, storage, and WebSockets | Can reject valid payloads or accept unsupported ones; check the [specific API and encoding](references/platform-apis.md#serialization-boundaries). |

export async function renderToR2(env: Env, version: VersionRow, origin: string) {

  // HTML comes from the current renderer, so it must point at the current client bundle.
  await requireSharedAssets(env);

/** KV allows one write per key per second; retry rate-limited writes a few times. */
export async function kvWrite(op: () => Promise<void>) {
  for (let attempt = 0; ; attempt++) {
    try {
      return await op();
    } catch (error) {
      const limited = /429|too many/i.test(String(error));
      if (!limited || attempt >= 3) throw error;
      await sleep(1100);
    }
  }
}

/**
 * Rewrites the KV record of every hostname of a site from D1, the source of truth: status,
 * published pointer and redirect. One writer per site at a time (D1 lease). A caller that
 * cannot take the lease leaves the site marked unsynced; the holder re-reads D1 after writing
 * and catches up, and the reconcile cron covers anything left. Returns whether KV now matches.
 */
export async function syncHostRecords(env: Env, siteId: string): Promise<boolean> {
  await clearRoutesSynced(env.DB, siteId);
  const token = await acquireRoutesLease(env.DB, siteId);
  if (!token) return false;
  try {
    for (let attempt = 0; attempt < 3; attempt++) {
      const state = await snapshot(env, siteId);
      for (const domain of state.domains) {
        // Fence: a holder whose lease expired must not overwrite a newer writer.
        if (!(await holdsRoutesLease(env.DB, siteId, token))) return false;
        const record = expectedRecord(state, domain);
        await kvWrite(() =>
          record
            ? env.ROUTES.put(kvHostKey(domain.hostname), JSON.stringify(record))
            : env.ROUTES.delete(kvHostKey(domain.hostname)),
        );
        // The lease may have expired while the write was in flight; then this write may be
        // stale, so leave the site dirty for the cron instead of trusting anyone's marker.
        if (!(await holdsRoutesLease(env.DB, siteId, token))) {
          await clearRoutesSynced(env.DB, siteId);
          return false;
        }
      }
      if ((await snapshot(env, siteId)).key === state.key) {
        await markRoutesSynced(env.DB, siteId, state.site.updated_at);
        return true;
      }
    }
    return false;
  } finally {
    await releaseRoutesLease(env.DB, siteId, token);
  }
}

/**
 * For request handlers: syncs now, retrying briefly while another writer holds the lease, so
 * the caller's change is live when it responds. The cron catches anything still pending.
 */
export async function syncHostRecordsNow(env: Env, siteId: string) {
  for (let attempt = 0; attempt < 4; attempt++) {
    if (await syncHostRecords(env, siteId)) return true;
    await sleep(1000);
  }
  return false;
}

/**
 * Cron safety net: re-syncs sites whose host records may lag D1 (a request failed after D1
 * committed, or KV kept rejecting writes). Guarantees serving converges to D1 within minutes.
 */
export async function reconcileHostRecords(env: Env) {
  const ids = await sitesNeedingRouteSync(env.DB);
  let synced = 0;
  for (const id of ids) {
    try {
      if (await syncHostRecords(env, id)) synced++;
    } catch (error) {
      console.error("[routes] reconcile failed", id, error);
    }
  }
  return { pending: ids.length, synced };
}

      if (known.has(hostname)) continue;

      const exists = await env.DB.prepare("SELECT 1 AS n FROM domains WHERE hostname = ?").bind(hostname).first();

      orphans++;

    const state = await snapshot(env, id);
    for (const domain of state.domains) {
      const expected = expectedRecord(state, domain);
      const actual = await env.ROUTES.get(kvHostKey(domain.hostname));
      if ((expected ? JSON.stringify(expected) : null) !== actual) {
        await clearRoutesSynced(env.DB, id);
        dirty++;
        break;
      }
    }
  }
  return { sites: results.length, dirty, orphans };
}

async function serveImmutable(env: Env, key: string) {
  const object = await env.SITES.get(key);
  if (!object) return new Response("not found", { status: 404 });
  const headers = new Headers();
  object.writeHttpMetadata(headers);
  headers.set("cache-control", "public, max-age=31536000, immutable");
  return new Response(object.body, { headers });
}

async function handle(request: Request, env: Env): Promise<Response> {

  // Uploaded photos (public, content-addressed) so previews show them before publishing.
  const media = method === "GET" ? MEDIA_PATH.exec(url.pathname) : null;

  When a deployment changes only server-rendered HTML, `ASSET_VERSION` can remain unchanged: `build-assets.mjs` hashes the generated CSS, client bundle, and images, not the server renderer.
