## Keep Compatibility Dates Current

Use today's date for new Workers. Encourage periodic updates for existing Workers, reviewing compatibility changes and running relevant tests. Assess existing behavior against its configured date and flags; see [compatibility guidance](references/configuration.md#keep-compatibility_date-current).

## Enable Observability

Enable [Workers Logs](https://developers.cloudflare.com/workers/observability/logs/workers-logs/index.md) and [Traces](https://developers.cloudflare.com/workers/observability/traces/index.md) when creating or preparing a Worker for production. Set `observability.enabled` and `observability.traces.enabled` to `true`; the top-level setting alone does not enable traces. Use structured JSON logging and configure sampling for the workload. During reviews, flag missing logs or traces. See the [configuration example](references/configuration.md#enable-workers-logs-and-traces).

## Anti-Patterns to Flag

| Anti-pattern | Consequence and preferred pattern |
|-------------|-----------------------------------|
| `await response.text()` or similar buffering on unbounded data | Can exhaust Worker memory; [stream large or unbounded bodies](references/runtime-patterns.md#stream-request-and-response-bodies). |
| Hardcoded secrets in source or config | Leaks credentials through version control; use Wrangler secrets. |
| `Math.random()` for security-sensitive tokens or IDs | Predictable values; use `crypto.randomUUID()` or `crypto.getRandomValues()`. |
| Async work started without awaiting, returning, or attaching it to `ctx.waitUntil()` | Work can be dropped and errors missed; tie it to the request or background-work lifetime. |
| Module-level mutable request state | Leaks data across requests and can cause I/O ownership errors; pass request state explicitly. |
| Cloudflare REST API calls for operations available through Worker bindings | Adds network and authentication overhead; use the available binding. |
| `ctx.passThroughOnException()` used as general error handling | Can conceal Worker failures by forwarding to the origin; use explicit error handling and structured error responses. |
| Hand-written `Env` that duplicates Wrangler bindings | Can drift from configuration; generate binding types with `wrangler types`. |
| Direct string comparison of secret values | Can expose timing differences; use the [Web Crypto comparison pattern](references/runtime-patterns.md#use-web-crypto-for-secure-token-generation). |
| Destructuring `ctx` methods, such as `const { waitUntil } = ctx` | Loses the receiver; call `ctx.waitUntil(...)`. |
| `any` on `Env` or handler parameters | Hides binding and handler contract errors; use the project's generated and platform types. |
| `as unknown as T` to force a platform type match | Hides incompatibilities; fix the underlying contract. |
| `implements` used in place of extending a platform base class | Does not inherit runtime behavior, `this.ctx`, or `this.env`; use the appropriate base class. |
| Unbound `env.X` in a platform class method | Bindings are available through `this.env.X`; see [binding access patterns](references/platform-apis.md#binding-access--the-most-common-error). |
| Applying one serialization rule across Queues, Workflow steps, storage, and WebSockets | Can reject valid payloads or accept unsupported ones; check the [specific API and encoding](references/platform-apis.md#serialization-boundaries). |

  // the blog feed stopped and routing marked for reconciliation.
  await env.DB.batch([
    env.DB.prepare("UPDATE sites SET cancelled_at = ?, status = 'suspended', updated_at = ?, routes_synced_at = NULL WHERE id = ?").bind(
      at,
      at,
      site.id,
    ),
    env.DB.prepare("UPDATE domains SET status = 'removing' WHERE site_id = ? AND kind = 'custom'").bind(site.id),
    env.DB.prepare("DELETE FROM blog_feeds WHERE site_id = ?").bind(site.id),
  ]);
  await syncHostRecordsNow(env, site.id);
  const domains: { hostname: string; removed: boolean; error?: string }[] = [];
  for (const { hostname } of results) {
    try {
      await removeDomain(env, site.id, hostname);
      domains.push({ hostname, removed: true });
    } catch (error) {
      // The row stays `removing`; the domains cron finishes it.
      domains.push({ hostname, removed: false, error: error instanceof Error ? error.message : String(error) });
    }
  }
  await env.ROUTES.delete(kvBlogKey(site.id));
  const purgeAfter = new Date(Date.parse(at) + retentionDays(env) * 86_400_000).toISOString();
  return { siteId: site.id, cancelledAt: at, purgeAfter, domains };
}

/** Undo a cancellation within the retention period. Custom domains must be connected again. */
export async function reactivateSite(env: Env, site: SiteRow) {
  if (!site.cancelled_at) throw new HttpError(409, "해지된 사이트가 아닙니다");
  const resumed = await env.DB.prepare(
    "UPDATE sites SET cancelled_at = NULL, status = 'active', updated_at = ?, routes_synced_at = NULL WHERE id = ? AND purging_at IS NULL",
  )
    .bind(now(), site.id)
    .run();
  if (resumed.meta.changes !== 1) throw new HttpError(409, "삭제가 진행 중인 사이트입니다");
  await syncHostRecordsNow(env, site.id);
  // Cancellation stopped the blog feed; the live version decides whether it comes back.
  if (site.published_version_id) {
    const live = await env.DB.prepare("SELECT config FROM site_versions WHERE id = ?").bind(site.published_version_id).first<{ config: string }>();
    if (live) await syncBlogFeed(env, site.id, JSON.parse(live.config) as { business?: { blogUrl?: string } });
  }
  return { siteId: site.id, status: "active" };
}

export async function listCancellations(env: Env) {
  const { results } = await env.DB.prepare("SELECT id, slug, cancelled_at FROM sites WHERE cancelled_at IS NOT NULL ORDER BY cancelled_at")
    .all<{ id: string; slug: string; cancelled_at: string }>();
  const days = retentionDays(env);
  return results.map((row) => {
    const purgeAfter = new Date(Date.parse(row.cancelled_at) + days * 86_400_000).toISOString();
    return { ...row, purgeAfter, purgeable: purgeAfter <= now() };
  });
}

/**
 * Permanent deletion after the retention period. Irreversible, so never run by cron: an
 * operator confirms with the site's slug. Removes R2 files, KV records and every D1 row.
 */
export async function purgeSite(env: Env, site: SiteRow, confirmSlug: unknown) {
  if (confirmSlug !== site.slug) throw new HttpError(422, "확인용 slug가 일치하지 않습니다");
  if (!site.cancelled_at) throw new HttpError(409, "해지된 사이트만 삭제할 수 있습니다");
  const purgeAfter = Date.parse(site.cancelled_at) + retentionDays(env) * 86_400_000;
  if (Date.now() < purgeAfter) {
    throw new HttpError(409, `보관 기간이 끝나지 않았습니다 (${new Date(purgeAfter).toISOString().slice(0, 10)} 이후 삭제 가능)`);
  }
  // Take the purge state first (conditional on the cancellation still standing); from here on
  // reactivation refuses, so nothing can resume the site while its data is being deleted.
  const claimed = await env.DB.prepare(
    "UPDATE sites SET purging_at = COALESCE(purging_at, ?) WHERE id = ? AND cancelled_at IS NOT NULL AND cancelled_at <= ?",
  )
    .bind(now(), site.id, new Date(Date.now() - retentionDays(env) * 86_400_000).toISOString())
    .run();
  if (claimed.meta.changes !== 1) throw new HttpError(409, "해지가 취소되었거나 보관 기간이 끝나지 않았습니다");
  const { results: domains } = await env.DB.prepare("SELECT hostname, kind, status FROM domains WHERE site_id = ?")
    .bind(site.id)
    .all<{ hostname: string; kind: string; status: string }>();
  if (domains.some((d) => d.kind === "custom")) {
    throw new HttpError(409, "커스텀 도메인 삭제가 끝나지 않았습니다. 잠시 뒤 다시 시도하세요");
  }
  const sweep = async () => {
    let deleted = 0;
    let cursor: string | undefined;
    do {
      const page = await env.SITES.list({ prefix: `sites/${site.id}/`, cursor });
      if (page.objects.length) await env.SITES.delete(page.objects.map((o) => o.key));
      deleted += page.objects.length;
      cursor = page.truncated ? page.cursor : undefined;
    } while (cursor);
    return deleted;
  };
  let objects = await sweep();
  for (const domain of domains) await env.ROUTES.delete(kvHostKey(domain.hostname));
  await env.ROUTES.delete(kvSiteKey(site.id));
  await env.ROUTES.delete(kvBlogKey(site.id));
  const tables = ["inquiries", "metrics_daily", "assets", "blog_feeds", "staged_renders", "domains", "site_versions"];
  await env.DB.batch([
    ...tables.map((table) => env.DB.prepare(`DELETE FROM ${table} WHERE site_id = ?`).bind(site.id)),
    env.DB.prepare("DELETE FROM sites WHERE id = ?").bind(site.id),
  ]);
  // Writers that started before the cancellation clean up after themselves (they re-check the
  // site after writing); a last sweep catches anything written between the first and the rows.
  objects += await sweep();
  return { siteId: site.id, slug: site.slug, objectsDeleted: objects, purged: true };
}

const toBytes = (text: string) => new TextEncoder().encode(text);

/** `/_shared/...` and `/_media/...` paths referenced by HTML (src, href, srcset). */
function referencedFiles(html: string): string[] {
  const out = new Set<string>();
  for (const match of html.matchAll(/(?:src|href|srcset|imagesrcset)="([^"]+)"/gi)) {
    for (const candidate of match[1]!.split(",")) {
      const path = candidate.trim().split(/\s+/)[0] ?? "";
      if (/^\/_(shared|media)\//.test(path)) out.add(path);
    }
  }
  return [...out];
}

const r2KeyFor = (file: string) =>
  file.startsWith("/_media/")
    ? `sites/${file.split("/")[2]}/media/${file.split("/").slice(3).join("/")}`
    : `shared/${file.slice("/_shared/".length)}`;

/**
 * Everything a customer can take with them: the live site as static files (HTML, the shared
 * assets it uses, uploaded photos) and their data (all versions, inquiries, daily metrics).
 * Streamed: each R2 object flows through the ZIP writer into the response, so memory stays
 * at one chunk however many photos a site has. Root-relative links: serve at a domain root.
 */
export async function exportSite(env: Env, siteId: string): Promise<{ filename: string; body: ReadableStream<Uint8Array> }> {
  const site = await getSite(env.DB, siteId);
  const { readable, writable } = new TransformStream<Uint8Array, Uint8Array>();
  const writer = writable.getWriter();
  const zip = new Zip((error, chunk, final) => {
    if (error) {
      void writer.abort(error);
      return;
    }
    void writer.write(chunk);
    if (final) void writer.close();
  });
  const addBytes = (name: string, bytes: Uint8Array) => {
    const entry = new ZipDeflate(name, { level: 6 });
    zip.add(entry);
    entry.push(bytes, true);
  };
  const addObject = async (name: string, object: R2ObjectBody) => {
    const entry = /\.(jpe?g|png|webp)$/.test(name) ? new ZipPassThrough(name) : new ZipDeflate(name, { level: 6 });
    zip.add(entry);
    const reader = object.body.getReader();
    for (;;) {
      // Backpressure: read the next chunk only when the download can take more. A client that
      // disconnects rejects `ready`, which stops production.
      await writer.ready;
      const { done, value } = await reader.read();
      if (done) break;
      entry.push(value);
    }
    entry.push(new Uint8Array(0), true);
  };

  const added = new Set<string>();
  const produce = async () => {
    if (site.published_version_id) {
      const version = await env.DB.prepare("SELECT id, number, config, render_id FROM site_versions WHERE id = ?")
        .bind(site.published_version_id)
        .first<{ id: string; number: number; config: string; render_id: string | null }>();
      if (version) {
        const assets = new Set<string>();
        for (const route of routesOf(JSON.parse(version.config) as { version?: unknown })) {
          const path = route === "/" ? "index.html" : `${route.slice(1)}/index.html`;
          const object = await env.SITES.get(r2PageKey(site.id, version.id, path, version.render_id));
          if (!object) continue;
          const html = await object.text();
          addBytes(`site/${path}`, toBytes(html));
          for (const file of referencedFiles(html)) assets.add(file);
        }
        // The client bundle loads chunks dynamically: take the whole _assets folder it lives in.
        const bundle = [...assets].find((p) => p.endsWith("/_assets/client.js"));
        if (bundle) {
          const listed = await env.SITES.list({ prefix: `shared/${bundle.slice("/_shared/".length, -"client.js".length)}` });
          for (const o of listed.objects) assets.add(`/_shared/${o.key.slice("shared/".length)}`);
        }
        for (const file of assets) {
          const object = await env.SITES.get(r2KeyFor(file));
          if (object) await addObject(`site${file}`, object);
          added.add(`site${file}`);
        }
      }
    }
    // Every uploaded photo, including ones only drafts, older versions or an unpublished site
    // use: the exported version data must stay recoverable after deletion.
    let cursor: string | undefined;
    do {
      const page = await env.SITES.list({ prefix: `sites/${site.id}/media/`, cursor });
      for (const o of page.objects) {
        const path = `site/_media/${site.id}/${o.key.slice(`sites/${site.id}/media/`.length)}`;
        if (added.has(path)) continue;
        const object = await env.SITES.get(o.key);
        if (object) await addObject(path, object);
        added.add(path);
      }
      cursor = page.truncated ? page.cursor : undefined;
    } while (cursor);
    const { results: versions } = await env.DB.prepare(
      "SELECT id, number, config, created_by, note, created_at, first_published_at FROM site_versions WHERE site_id = ? ORDER BY number",
    )
      .bind(site.id)
      .all<{ config: string }>();
    addBytes("data/versions.json", toBytes(JSON.stringify(versions.map((v) => ({ ...v, config: JSON.parse(v.config) })), null, 2)));
    addBytes(
      "data/site.json",
      toBytes(JSON.stringify({ id: site.id, slug: site.slug, status: site.status, cancelled_at: site.cancelled_at, created_at: site.created_at }, null, 2)),
    );
    addBytes("data/inquiries.csv", toBytes(inquiriesCsv(await allInquiries(env, site.id, {}))));
    const { results: metrics } = await env.DB.prepare("SELECT date, event, count FROM metrics_daily WHERE site_id = ? ORDER BY date, event")
      .bind(site.id)
      .all<{ date: string; event: string; count: number }>();
    addBytes("data/metrics.csv", toBytes(`\uFEFFdate,event,count\r\n${metrics.map((m) => `${m.date},${m.event},${m.count}`).join("\r\n")}\r\n`));
    addBytes(
      "README.txt",
      toBytes(
        [
          `${site.slug} 홈페이지 내보내기 (${now().slice(0, 10)})`,
          "",
          "site/  게시 중인 홈페이지 파일입니다. 웹 서버의 도메인 루트(/)에 그대로 올리면 열립니다.",
          "data/versions.json  모든 버전의 사이트 설정(문구·사진 키·연락처)",
          "data/inquiries.csv  홈페이지 문의 기록 (엑셀에서 열 수 있습니다)",
          "data/metrics.csv  일별 방문·연락 클릭 수",
        ].join("\r\n"),
      ),
    );
    zip.end();
  };
  // Runs while the response streams; a failure aborts the download instead of a truncated zip.
  produce().catch((error) => void writer.abort(error));
  return { filename: `${site.slug}-export.zip`, body: readable };
}

const sleep = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));

/** KV allows one write per key per second; retry rate-limited writes a few times. */
export async function kvWrite(op: () => Promise<void>) {
  for (let attempt = 0; ; attempt++) {
    try {
      return await op();
    } catch (error) {
      const limited = /429|too many/i.test(String(error));
      if (!limited || attempt >= 3) throw error;
      await sleep(1100);
    }
  }
}

/** Public status of a site as the router sees it. */
export function hostStatus(site: SiteRow): HostRecord["status"] {
  if (site.cancelled_at) return "cancelled";
  return site.status === "suspended" ? "suspended" : "active";
}

type Snapshot = Awaited<ReturnType<typeof snapshot>>;
type DomainRow = Snapshot["domains"][number];

/** What KV should hold for a hostname; null means the key must not exist. */
function expectedRecord(state: Snapshot, domain: DomainRow): HostRecord | null {
  // Custom hostnames that are not active yet must not serve the site.
  if (domain.kind === "custom" && domain.status !== "active") return null;
  return {
    siteId: state.site.id,
    status: state.status,
    pointer: state.pointer,
    redirectTo: domain.hostname === state.primary ? null : state.primary,
  };
}

/** Routes a stored config renders to (v2 adds the privacy page). */
export const routesOf = (config: { version?: unknown }) => (config.version === 2 ? ["/", "/privacy"] : ["/"]);

async function snapshot(env: Env, siteId: string) {
  const { site, version } = await servingState(env.DB, siteId);
  const pointer: PublishedPointer | null =
    version
      ? {
          versionId: version.id,
          number: version.number,
          // Rows from before migration 0003 are backfilled; never drop a live pointer over a missing time.
          publishedAt: site.published_at ?? site.updated_at,
          pages: routesOf(JSON.parse(version.config) as { version?: unknown }),
          renderId: version.render_id,
        }
      : null;
  const primary = await primaryHostname(env.DB, site, env.BASE_DOMAIN);
  const { results: domains } = await env.DB.prepare("SELECT hostname, kind, status FROM domains WHERE site_id = ?")
    .bind(site.id)
    .all<{ hostname: string; kind: string; status: string }>();
  const status = hostStatus(site);
  const key = JSON.stringify([status, pointer?.versionId, pointer?.publishedAt, pointer?.renderId, primary, domains]);
  return { site, pointer, primary, domains, status, key };
}

/**
 * Rewrites the KV record of every hostname of a site from D1, the source of truth: status,
 * published pointer and redirect. One writer per site at a time (D1 lease). A caller that
 * cannot take the lease leaves the site marked unsynced; the holder re-reads D1 after writing
 * and catches up, and the reconcile cron covers anything left. Returns whether KV now matches.
 */
export async function syncHostRecords(env: Env, siteId: string): Promise<boolean> {
  await clearRoutesSynced(env.DB, siteId);
  const token = await acquireRoutesLease(env.DB, siteId);
  if (!token) return false;
  try {
    for (let attempt = 0; attempt < 3; attempt++) {
      const state = await snapshot(env, siteId);
      for (const domain of state.domains) {
        // Fence: a holder whose lease expired must not overwrite a newer writer.
        if (!(await holdsRoutesLease(env.DB, siteId, token))) return false;
        const record = expectedRecord(state, domain);
        await kvWrite(() =>
          record
            ? env.ROUTES.put(kvHostKey(domain.hostname), JSON.stringify(record))
            : env.ROUTES.delete(kvHostKey(domain.hostname)),
        );
        // The lease may have expired while the write was in flight; then this write may be
        // stale, so leave the site dirty for the cron instead of trusting anyone's marker.
        if (!(await holdsRoutesLease(env.DB, siteId, token))) {
          await clearRoutesSynced(env.DB, siteId);
          return false;
        }
      }
      if ((await snapshot(env, siteId)).key === state.key) {
        await markRoutesSynced(env.DB, siteId, state.site.updated_at);
        return true;
      }
    }
    return false;
  } finally {
    await releaseRoutesLease(env.DB, siteId, token);
  }
}

/**
 * For request handlers: syncs now, retrying briefly while another writer holds the lease, so
 * the caller's change is live when it responds. The cron catches anything still pending.
 */
export async function syncHostRecordsNow(env: Env, siteId: string) {
  for (let attempt = 0; attempt < 4; attempt++) {
    if (await syncHostRecords(env, siteId)) return true;
    await sleep(1000);
  }
  return false;
}

/**
 * Cron safety net: re-syncs sites whose host records may lag D1 (a request failed after D1
 * committed, or KV kept rejecting writes). Guarantees serving converges to D1 within minutes.
 */
export async function reconcileHostRecords(env: Env) {
  const ids = await sitesNeedingRouteSync(env.DB);
  let synced = 0;
  for (const id of ids) {
    try {
      if (await syncHostRecords(env, id)) synced++;
    } catch (error) {
      console.error("[routes] reconcile failed", id, error);
    }
  }
  return { pending: ids.length, synced };
}

/**
 * Hourly audit: compares every site's expected host records with KV and marks mismatches
 * dirty, so the reconcile pass repairs them whatever race produced them. Writes nothing itself.
 */
export async function auditHostRecords(env: Env) {
  const { results } = await env.DB.prepare("SELECT id FROM sites").all<{ id: string }>();
  let dirty = 0;
  // Orphans: host records whose hostname no longer exists in D1 (a race recreated one after
  // removal). Routing must never outlive the domain row, so they are deleted here.
  const known = new Set(
    (await env.DB.prepare("SELECT hostname FROM domains").all<{ hostname: string }>()).results.map((row) => row.hostname),
  );
  let orphans = 0;
  let cursor: string | undefined;
  do {
    const page = await env.ROUTES.list({ prefix: "host:", cursor });
    for (const key of page.keys) {
      const hostname = key.name.slice("host:".length);
      if (known.has(hostname)) continue;
      // The snapshot may predate a site created during this audit: confirm against D1 now.
      const exists = await env.DB.prepare("SELECT 1 AS n FROM domains WHERE hostname = ?").bind(hostname).first();
      if (exists) continue;
      await kvWrite(() => env.ROUTES.delete(key.name));
      orphans++;
    }
    cursor = page.list_complete ? undefined : page.cursor;
  } while (cursor);
  for (const { id } of results) {
    const state = await snapshot(env, id);
    for (const domain of state.domains) {
      const expected = expectedRecord(state, domain);
      const actual = await env.ROUTES.get(kvHostKey(domain.hostname));
      if ((expected ? JSON.stringify(expected) : null) !== actual) {
        await clearRoutesSynced(env.DB, id);
        dirty++;
        break;
      }
    }
  }
  return { sites: results.length, dirty, orphans };
}

/** Everything outside the config that changes the HTML (stored in rendered_origin). */
export const renderKeyOf = (env: Env, origin: string) => JSON.stringify({ origin, turnstileFrame: turnstileFrameUrl(env) });

/**
 * Renders every page of a version with the current renderer and shared assets under a fresh,
 * immutable render_id. Nothing points at it yet: concurrent renders never mix pages.
 */
export async function renderToR2(env: Env, version: VersionRow, origin: string) {
  const config = JSON.parse(version.config) as SiteConfig;
  // HTML comes from the current renderer, so it must point at the current client bundle.
  await requireSharedAssets(env);
  const { pages } = await renderSite(config, { assetBase: assetBase(), origin, turnstileFrame: turnstileFrameUrl(env) });
  const renderId = crypto.randomUUID().replace(/-/g, "").slice(0, 16);
  const written: string[] = [];
  for (const page of pages) {
    const key = r2PageKey(version.site_id, version.id, page.path, renderId);
    written.push(key);
    await env.SITES.put(key, page.html, { httpMetadata: { contentType: "text/html; charset=utf-8" } });
  }
  // Any render (publish, rollback, domain maintenance, republish) cleans up if a purge started.
  await ensureNotPurged(env, version.site_id, written);
  return { renderId, renderKey: renderKeyOf(env, origin), routes: pages.map((page) => page.route) };
}

/**
 * Renders a version unless it is already rendered against the current shared assets and
 * origin, then records the render (the pointer follows it on the next sync).
 */
async function ensureRendered(env: Env, version: VersionRow, origin: string, { keepAssets = false } = {}) {
  const config = JSON.parse(version.config) as SiteConfig;
  // Rollback keeps the version's own render; publish moves it to the current renderer and assets.
  // (site_versions.asset_version holds RENDER_VERSION: assets plus server renderer.)
  const assetsOk = keepAssets || version.asset_version === RENDER_VERSION;
  if (version.rendered_at && assetsOk && version.rendered_origin === renderKeyOf(env, origin)) {
    return routesOf(config);
  }
  const render = await renderToR2(env, version, origin);
  await markRendered(env.DB, version.id, RENDER_VERSION, render.renderKey, render.renderId);
  return render.routes;
}

async function pointTo(env: Env, site: SiteRow, version: VersionRow, pages: string[]) {
  const publishedAt = await markPublished(env.DB, site.id, version.id, RENDER_VERSION);
  const renderId = (await getVersion(env.DB, site.id, version.id)).render_id;
  const pointer: PublishedPointer = { versionId: version.id, number: version.number, publishedAt, pages, renderId };
  await env.ROUTES.put(kvSiteKey(site.id), JSON.stringify(pointer));
  await syncHostRecordsNow(env, site.id);
  await syncBlogFeed(env, site.id, JSON.parse(version.config) as { business?: { blogUrl?: string } });
  return pointer;
}

/** Publish = render (if needed) → R2 → move the KV pointer. Visible worldwide within about a minute. */
export async function publish(env: Env, site: SiteRow, versionId?: string) {
  if (site.cancelled_at) throw new HttpError(409, "해지된 사이트입니다. 먼저 해지를 취소하세요");
  await requireSharedAssets(env);
  const version = versionId ? await getVersion(env.DB, site.id, versionId) : await latestVersion(env.DB, site.id);
  const pages = await ensureRendered(env, version, await siteOrigin(env, site));
  return pointTo(env, site, version, pages);
}

/**
 * Rollback = move the pointer back to a version that was published before. No re-render, unless
 * the site's origin changed since that version was rendered (custom domain added or removed):
 * then only its HTML is re-rendered so canonical and OG URLs point at the current host.
 */
export async function rollback(env: Env, site: SiteRow, versionId: string) {
  if (site.cancelled_at) throw new HttpError(409, "해지된 사이트입니다. 먼저 해지를 취소하세요");
  const version = await getVersion(env.DB, site.id, versionId);
  if (!version.first_published_at) throw new HttpError(409, "게시된 적 없는 버전으로는 롤백할 수 없습니다");
  const exists = await env.SITES.head(r2PageKey(site.id, version.id, "index.html", version.render_id));
  if (!exists) throw new HttpError(409, "렌더된 파일이 R2에 없습니다. publish로 다시 게시하세요");
  const pages = await ensureRendered(env, version, await siteOrigin(env, site), { keepAssets: true });
  return pointTo(env, site, version, pages);
}

/**
 * The site's origin changed (custom domain activated, failed or removed): re-render the live
 * version's HTML in place (same R2 keys). The KV pointer is not touched, so this can never undo
 * a concurrent publish; if the live version changes meanwhile, the new one is checked too.
 * Returns false when it could not settle (the caller keeps origin_pending for the cron).
 */
export async function rerenderLive(env: Env, siteId: string): Promise<boolean> {
  for (let attempt = 0; attempt < 3; attempt++) {
    const site = await getSite(env.DB, siteId);
    if (!site.published_version_id) {
      await syncHostRecordsNow(env, siteId);
      return true;
    }
    const version = await getVersion(env.DB, siteId, site.published_version_id);
    await ensureRendered(env, version, await siteOrigin(env, site));
    // Settled only if the live version is unchanged and its HTML matches the origin as of now
    // (an overlapping domain change may have moved it while this render ran).
    const after = await getSite(env.DB, siteId);
    const rendered = await getVersion(env.DB, siteId, version.id);
    if (after.published_version_id === version.id && rendered.rendered_origin === renderKeyOf(env, await siteOrigin(env, after))) {
      await syncHostRecordsNow(env, siteId);
      return true;
    }
  }
  return false;
}

/**
 * Moves the publish pointer in D1. Refused (409) if the site was cancelled meanwhile: the
 * check is part of the same statements, not the caller's earlier snapshot.
 */
export async function markPublished(db: D1Database, siteId: string, versionId: string, templateVersion: string) {
  const at = now();
  const live = "EXISTS (SELECT 1 FROM sites WHERE id = ? AND cancelled_at IS NULL)";
  const [, pointed] = await db.batch([
    db
      .prepare(`UPDATE site_versions SET first_published_at = COALESCE(first_published_at, ?) WHERE id = ? AND ${live}`)
      .bind(at, versionId, siteId),
    db
      .prepare(
        "UPDATE sites SET published_version_id = ?, published_at = ?, template_version = ?, updated_at = ? WHERE id = ? AND cancelled_at IS NULL",
      )
      .bind(versionId, at, templateVersion, at, siteId),
  ]);
  if (pointed?.meta.changes !== 1) throw new HttpError(409, "해지된 사이트입니다. 먼저 해지를 취소하세요");
  return at;
}

/** Takes the per-site route-sync lease; null when another writer holds it. */
export async function acquireRoutesLease(db: D1Database, siteId: string): Promise<string | null> {
  const token = crypto.randomUUID();
  const now = new Date();
  const result = await db
    .prepare(
      "UPDATE sites SET routes_lock_token = ?, routes_lock_until = ? WHERE id = ? AND (routes_lock_until IS NULL OR routes_lock_until < ?)",
    )
    .bind(token, new Date(now.getTime() + LEASE_MS).toISOString(), siteId, now.toISOString())
    .run();
  return result.meta.changes === 1 ? token : null;
}

async function serveImmutable(env: Env, key: string) {
  const object = await env.SITES.get(key);
  if (!object) return new Response("not found", { status: 404 });
  const headers = new Headers();
  object.writeHttpMetadata(headers);
  headers.set("cache-control", "public, max-age=31536000, immutable");
  return new Response(object.body, { headers });
}

async function handle(request: Request, env: Env): Promise<Response> {
  const url = new URL(request.url);
  const parts = url.pathname.split("/").filter(Boolean);
  const method = request.method;

  if (method === "GET" && url.pathname === "/health") return json({ ok: true });

  // Shared assets are public and immutable; previews load them from here.
  if (method === "GET" && parts[0] === "_shared") return serveImmutable(env, r2SharedKey(parts.slice(1).join("/")));
  // Uploaded photos (public, content-addressed) so previews show them before publishing.
  const media = method === "GET" ? MEDIA_PATH.exec(url.pathname) : null;
  if (media) return serveImmutable(env, r2MediaKey(media[1]!, media[2]!));

  // Turnstile widget frame embedded by public sites (see turnstile.ts).
  if (method === "GET" && url.pathname === "/turnstile-frame") {
    if (!env.TURNSTILE_SITE_KEY) return new Response("turnstile not configured", { status: 404 });
    const parent = url.searchParams.get("parent") ?? "";
    if (!/^https?:\/\/[a-z0-9.-]+(:\d+)?$/i.test(parent)) return new Response("bad parent", { status: 400 });
    return new Response(turnstileFrame(env.TURNSTILE_SITE_KEY, parent), {
      headers: { "content-type": "text/html; charset=utf-8", "cache-control": "no-store", "x-robots-tag": "noindex" },
    });
  }

  // Preview: anyone with the token can view a version. Never indexed. /preview/<token>[/privacy]
  if (method === "GET" && parts[0] === "preview" && (parts.length === 2 || parts.length === 3)) {
    const version = await versionByPreviewToken(env.DB, parts[1]!);
    const site = await getSite(env.DB, version.site_id);
    const { pages } = await renderSite(JSON.parse(version.config) as SiteConfig, {
      assetBase: assetBase(),
      origin: await siteOrigin(env, site),
      preview: true,
      turnstileFrame: null,
      // Site links are root-relative (`/#inquiry`, `/privacy`); keep them inside this preview.
      linkPrefix: `/preview/${parts[1]}`,
    });
    const route = parts[2] ? `/${parts[2]}` : "/";
    const page = pages.find((entry) => entry.route === route);
    if (!page) throw new HttpError(404, "없는 페이지입니다");
    return new Response(page.html, {

/**
 * Cron: finish interrupted removals, retry pending origin re-renders, check pending hostnames
 * every few minutes (for 14 days) and active ones every 6 hours (a moved or expired hostname
 * falls back to the platform subdomain).
 */
export async function maintainDomains(env: Env) {
  const result = { removed: 0, rerendered: 0, checked: 0, changed: 0 };
  const { results: removing } = await env.DB.prepare("SELECT hostname, site_id FROM domains WHERE status = 'removing' LIMIT 20")
    .all<{ hostname: string; site_id: string }>();
  for (const row of removing) {
    try {
      await removeDomain(env, row.site_id, row.hostname);
      result.removed++;
    } catch (error) {
      console.error("[domains] removal retry failed", row.hostname, error);
    }
  }
  const { results: pendingOrigins } = await env.DB.prepare("SELECT id FROM sites WHERE origin_pending = 1 LIMIT 20").all<{ id: string }>();
  for (const { id } of pendingOrigins) {
    try {
      await afterPrimaryChange(env, id);
      result.rerendered++;
    } catch (error) {
      console.error("[domains] origin re-render failed", id, error);
    }
  }
  if (!SaasClient.configured(env)) return result;
  const at = Date.now();
  const { results } = await env.DB.prepare(
    `SELECT * FROM domains WHERE kind = 'custom' AND (
       (status = 'pending' AND activated_at IS NULL AND created_at >= ? AND (checked_at IS NULL OR checked_at < ?))
       OR (status IN ('pending', 'failed') AND activated_at IS NOT NULL AND (checked_at IS NULL OR checked_at < ?))
       OR (status = 'active' AND (checked_at IS NULL OR checked_at < ?))
     ) ORDER BY checked_at LIMIT 50`,
  )
    .bind(
      new Date(at - 14 * 86_400_000).toISOString(),
      new Date(at - 4 * 60_000).toISOString(),
      new Date(at - 30 * 60_000).toISOString(),
      new Date(at - 6 * 3_600_000).toISOString(),
    )
    .all<DomainRow>();
  for (const row of results) {
    try {
      if ((await refreshDomain(env, row)) !== row.status) result.changed++;
      result.checked++;
    } catch (error) {
      console.error("[domains] refresh failed", row.hostname, error);
    }
  }
  return result;
}

/**
 * Removal is resumable: stop billing at Cloudflare, mark the row `removing` (routing treats it
 * as gone), restore the primary host, and only then delete the row. A failure at any step
 * leaves a `removing` row that a retry or the cron finishes.
 */
export async function removeDomain(env: Env, siteId: string, hostname: string) {
  const row = await env.DB.prepare("SELECT * FROM domains WHERE hostname = ? AND site_id = ? AND kind = 'custom'")
    .bind(hostname, siteId)
    .first<DomainRow>();
  if (!row) throw new HttpError(404, "연결된 커스텀 도메인이 아닙니다");
  // Intent first, together with the routing and origin retry markers: if anything below stops
  // (even a Cloudflare outage), the cron restores the platform host and finishes the removal.
  if (row.status !== "removing") {
    await env.DB.batch([
      env.DB.prepare("UPDATE domains SET status = 'removing' WHERE hostname = ?").bind(hostname),
      env.DB.prepare("UPDATE sites SET origin_pending = 1, routes_synced_at = NULL, updated_at = ? WHERE id = ?").bind(now(), siteId),
    ]);
  }
  if (row.cf_hostname_id) await requireSaas(env).delete(row.cf_hostname_id);
  await afterPrimaryChange(env, siteId);
  // Final cleanup under the route-writer lease, so no in-flight sync can recreate the record.
  const token = await acquireRoutesLease(env.DB, siteId);
  if (!token) throw new HttpError(409, "라우팅을 갱신하는 중입니다. 삭제는 곧 자동으로 마무리됩니다");
  try {
    await kvWrite(() => env.ROUTES.delete(kvHostKey(hostname)));
    await env.DB.prepare("DELETE FROM domains WHERE hostname = ? AND status = 'removing'").bind(hostname).run();
  } finally {
    await releaseRoutesLease(env.DB, siteId, token);
  }
  await touchSite(env, siteId);
  return { hostname, removed: true };
}

/** Lowercase FQDN without scheme, path or port; rejects platform hosts. */
export function normalizeHostname(input: unknown, baseDomain: string): string {
  if (typeof input !== "string") throw new HttpError(422, "hostname이 필요합니다");
  const host = input.trim().toLowerCase().replace(/^https?:\/\//, "").replace(/\/.*$/, "").replace(/\.$/, "");
  const label = "[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?";
  if (!new RegExp(`^(?:${label}\\.)+[a-z]{2,63}$`).test(host) || host.length > 253) {
    throw new HttpError(422, `도메인 형식이 아닙니다: ${input}`);
  }
  const platform = [baseDomain, PLATFORM_ZONE];
  if (platform.some((zone) => host === zone || host.endsWith(`.${zone}`)) || host.endsWith(".localhost")) {
    throw new HttpError(422, "플랫폼 도메인은 연결할 수 없습니다");
  }
  return host;
}

/** `example.co.kr` and `example.com` are apex; `www.example.co.kr` is not. */
export function isApex(hostname: string) {
  const suffix = SUFFIXES.filter((s) => hostname.endsWith(`.${s}`)).sort((a, b) => b.length - a.length)[0];
  if (!suffix) return hostname.split(".").length === 2;
  return hostname.slice(0, -(suffix.length + 1)).split(".").length === 1;
}
