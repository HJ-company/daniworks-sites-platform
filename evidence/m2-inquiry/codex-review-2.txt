## Keep Compatibility Dates Current

Use today's date for new Workers. Encourage periodic updates for existing Workers, reviewing compatibility changes and running relevant tests. Assess existing behavior against its configured date and flags; see [compatibility guidance](references/configuration.md#keep-compatibility_date-current).

## Enable Observability

Enable [Workers Logs](https://developers.cloudflare.com/workers/observability/logs/workers-logs/index.md) and [Traces](https://developers.cloudflare.com/workers/observability/traces/index.md) when creating or preparing a Worker for production. Set `observability.enabled` and `observability.traces.enabled` to `true`; the top-level setting alone does not enable traces. Use structured JSON logging and configure sampling for the workload. During reviews, flag missing logs or traces. See the [configuration example](references/configuration.md#enable-workers-logs-and-traces).

## Anti-Patterns to Flag

| Anti-pattern | Consequence and preferred pattern |
|-------------|-----------------------------------|
| `await response.text()` or similar buffering on unbounded data | Can exhaust Worker memory; [stream large or unbounded bodies](references/runtime-patterns.md#stream-request-and-response-bodies). |
| Hardcoded secrets in source or config | Leaks credentials through version control; use Wrangler secrets. |
| `Math.random()` for security-sensitive tokens or IDs | Predictable values; use `crypto.randomUUID()` or `crypto.getRandomValues()`. |
| Async work started without awaiting, returning, or attaching it to `ctx.waitUntil()` | Work can be dropped and errors missed; tie it to the request or background-work lifetime. |
| Module-level mutable request state | Leaks data across requests and can cause I/O ownership errors; pass request state explicitly. |
| Cloudflare REST API calls for operations available through Worker bindings | Adds network and authentication overhead; use the available binding. |
| `ctx.passThroughOnException()` used as general error handling | Can conceal Worker failures by forwarding to the origin; use explicit error handling and structured error responses. |
| Hand-written `Env` that duplicates Wrangler bindings | Can drift from configuration; generate binding types with `wrangler types`. |
| Direct string comparison of secret values | Can expose timing differences; use the [Web Crypto comparison pattern](references/runtime-patterns.md#use-web-crypto-for-secure-token-generation). |
| Destructuring `ctx` methods, such as `const { waitUntil } = ctx` | Loses the receiver; call `ctx.waitUntil(...)`. |
| `any` on `Env` or handler parameters | Hides binding and handler contract errors; use the project's generated and platform types. |
| `as unknown as T` to force a platform type match | Hides incompatibilities; fix the underlying contract. |
| `implements` used in place of extending a platform base class | Does not inherit runtime behavior, `this.ctx`, or `this.env`; use the appropriate base class. |
| Unbound `env.X` in a platform class method | Bindings are available through `this.env.X`; see [binding access patterns](references/platform-apis.md#binding-access--the-most-common-error). |
| Applying one serialization rule across Queues, Workflow steps, storage, and WebSockets | Can reject valid payloads or accept unsupported ones; check the [specific API and encoding](references/platform-apis.md#serialization-boundaries). |

async function siteName(env: Env, siteId: string) {
  const row = await env.DB.prepare(
    "SELECT s.slug, v.config FROM sites s LEFT JOIN site_versions v ON v.id = s.published_version_id WHERE s.id = ?",
  )
    .bind(siteId)

/**
 * Handles a visitor inquiry routed from the public router (via RPC, never from the internet):
 * validation, spam checks (honeypot, timing, Turnstile, rate limits), storage, then mail.
 */
export async function submitInquiry(
  env: Env,
  request: InquiryRequest,
  waitUntil: (promise: Promise<unknown>) => void,
): Promise<InquiryResponse> {
  const parsed = inquirySubmissionSchema.safeParse(request.body);
  if (!parsed.success) {
    return { status: 422, body: { ok: false, error: "입력 내용을 확인해 주세요", details: formatIssues(parsed.error) } };
  }
  const input = parsed.data;
  // Bots: answer like a success so they learn nothing, store nothing.
  if (input.website || (input.elapsedMs !== undefined && input.elapsedMs < LIMITS.minElapsedMs)) {
    return { status: 200, body: { ok: true } };
  }

  const site = await env.DB.prepare("SELECT id, status, cancelled_at, notify_email FROM sites WHERE id = ?")
    .bind(request.siteId)
    .first<{ id: string; status: string; cancelled_at: string | null; notify_email: string | null }>();
  if (!site || site.status !== "active" || site.cancelled_at) {
    return { status: 404, body: { ok: false, error: "문의를 받을 수 없는 사이트입니다" } };
  }

  const turnstile = await verifyTurnstile(env.TURNSTILE_SECRET, input.turnstileToken, request.ip);
  if (!turnstile.ok) {
    return { status: 403, body: { ok: false, error: "스팸 방지 확인에 실패했습니다. 다시 시도해 주세요." } };
  }

  const now = new Date();
  const day = now.toISOString().slice(0, 10);
  const clientHash = request.ip ? await sha256Hex(new TextEncoder().encode(`${env.HASH_PEPPER ?? "dsites"}:${day}:${request.ip}`), 32) : null;
  const windowStart = new Date(now.getTime() - 10 * 60_000).toISOString();
  const dayStart = `${day}T00:00:00.000Z`;
  const id = crypto.randomUUID();
  const createdAt = now.toISOString();
  const notify = site.notify_email ? "pending" : "skipped";
  // Limit checks and insert in one statement: D1 runs statements one at a time, so concurrent
  // submissions cannot all pass the counts before any of them is stored.
  const inserted = await env.DB.prepare(
    `INSERT INTO inquiries (id, site_id, name, phone, message, page, client_hash, notify_status, created_at)
     SELECT ?, ?, ?, ?, ?, ?, ?, ?, ?
     WHERE (? IS NULL OR (SELECT COUNT(*) FROM inquiries WHERE client_hash = ? AND created_at >= ?) < ?)
       AND (SELECT COUNT(*) FROM inquiries WHERE site_id = ? AND created_at >= ?) < ?`,
  )
    .bind(
      id, site.id, input.name, input.phone, input.message, input.page ?? null, clientHash, notify, createdAt,
      clientHash, clientHash, windowStart, LIMITS.perClient,
      site.id, dayStart, LIMITS.perSiteDay,
    )
    .run();
  if (inserted.meta.changes !== 1) {
    const siteFull =
      ((await env.DB.prepare("SELECT COUNT(*) AS n FROM inquiries WHERE site_id = ? AND created_at >= ?")
        .bind(site.id, dayStart)
        .first<{ n: number }>())?.n ?? 0) >= LIMITS.perSiteDay;
    return {
      status: 429,
      body: {
        ok: false,
        error: siteFull ? "오늘은 더 이상 문의를 받을 수 없습니다. 전화로 연락해 주세요." : "잠시 뒤 다시 시도해 주세요",
      },
    };
  }
  await bumpMetric(env.DB, site.id, "inquiry", now);

  if (site.notify_email) {
    waitUntil(notifyOwner(env, site.id, site.notify_email, { id, ...input, createdAt, hostname: request.hostname }));
  }
  return { status: 201, body: { ok: true, id } };
}

export function requireEmail(value: unknown): string | null {
  if (value === null || value === "") return null;
  if (typeof value !== "string" || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(value) || value.length > 200) {
    throw new HttpError(422, "알림 이메일 형식이 아닙니다");
  }
  return value.trim();
}

/**
 * Sends a transactional mail. `binding`: Cloudflare Email Service (`send_email` binding, the
 * sender domain must be onboarded). `mailpit`: local inbox for verification. `off`: skipped.
 */
export async function sendMail(env: Env, mail: OutgoingMail): Promise<{ id: string | null }> {
  const mode = env.MAIL_MODE ?? "off";
  const from = { email: env.MAIL_FROM ?? "noreply@dsites.kr", name: "홈페이지 알림" };
  if (mode === "binding") {
    if (!env.EMAIL) throw new Error("send_email 바인딩(EMAIL)이 없습니다");
    const result = await env.EMAIL.send({ from, to: mail.to, subject: mail.subject, text: mail.text, html: mail.html });
    return { id: result.messageId };
  }
  if (mode === "mailpit") {
    const res = await fetch(`${env.MAIL_PIT_URL ?? "http://127.0.0.1:8025"}/api/v1/send`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({
        From: { Email: from.email, Name: from.name },
        To: [{ Email: mail.to }],
        Subject: mail.subject,
        Text: mail.text,
        HTML: mail.html,
      }),
    });
    if (!res.ok) throw new Error(`mailpit ${res.status}: ${await res.text()}`);
    const body = (await res.json()) as { ID?: string };
    return { id: body.ID ?? null };
  }
  return { id: null };
}

const SITEVERIFY = "https://challenges.cloudflare.com/turnstile/v0/siteverify";

/** Server-side Turnstile check. A missing secret means Turnstile is not configured (allowed). */
export async function verifyTurnstile(
  secret: string | undefined,
  token: string | undefined,
  ip: string | null,
  fetcher: typeof fetch = fetch,
): Promise<{ ok: boolean; reason?: string }> {
  if (!secret) return { ok: true, reason: "not-configured" };
  if (!token) return { ok: false, reason: "missing-token" };
  const form = new FormData();
  form.append("secret", secret);
  form.append("response", token);
  if (ip) form.append("remoteip", ip);
  try {
    const res = await fetcher(SITEVERIFY, { method: "POST", body: form });
    const body = (await res.json()) as { success?: boolean; "error-codes"?: string[] };
    return body.success ? { ok: true } : { ok: false, reason: body["error-codes"]?.join(",") ?? "failed" };
  } catch (error) {
    return { ok: false, reason: `siteverify-unreachable: ${String(error)}` };
  }
}

export function turnstileFrame(siteKey: string, parentOrigin: string) {

export async function sha256Hex(bytes: Uint8Array, length = 20) {
  const digest = await crypto.subtle.digest("SHA-256", bytes as Uint8Array<ArrayBuffer>);
  return [...new Uint8Array(digest)].map((b) => b.toString(16).padStart(2, "0")).join("").slice(0, length);
}

/**
 * Turnstile token from the platform-hosted frame. The frame loads only after the visitor starts
 * filling the form, so the challenge script never weighs on page load.
 */
function useTurnstile(frameUrl: string | null, active: boolean) {
  const [token, setToken] = useState<string | null>(null);
  const [failed, setFailed] = useState(false);
  const [round, setRound] = useState(0);
  useEffect(() => {
    if (!frameUrl || !active) return;
    const frameOrigin = new URL(frameUrl, location.href).origin;
    function onMessage(event: MessageEvent) {
      if (event.origin !== frameOrigin || event.data?.type !== "dsites-turnstile") return;
      if (event.data.error) setFailed(true);
      else setToken(event.data.token ?? null);
    }
    window.addEventListener("message", onMessage);
    return () => window.removeEventListener("message", onMessage);
  }, [frameUrl, active]);
  const src = frameUrl && active ? `${frameUrl}?parent=${encodeURIComponent(location.origin)}&round=${round}` : null;
  // Tokens are single-use: reloading the frame issues a fresh one.
  const reset = () => {
    setToken(null);
    setFailed(false);
    setRound((n) => n + 1);
  };
  return { token, failed, src, required: Boolean(frameUrl), reset };
}

async function serveImmutable(env: Env, key: string) {
  const object = await env.SITES.get(key);
  if (!object) return new Response("not found", { status: 404 });
  const headers = new Headers();
  object.writeHttpMetadata(headers);
  headers.set("cache-control", "public, max-age=31536000, immutable");
  return new Response(object.body, { headers });
}

/** RPC surface of the api Worker's `PublicApi` entrypoint (apps/api/src/public.ts). */
interface PublicApi {
  submitInquiry(request: { siteId: string; hostname: string; ip: string | null; body: unknown }): Promise<{
    status: number;
    body: unknown;
  }>;
}

interface Env {
  SITES: R2Bucket;
  ROUTES: KVNamespace;
  BASE_DOMAIN: string;
  API: PublicApi;

const MAX_FORM_BYTES = 16 * 1024;

const jsonResponse = (status: number, body: unknown) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { "content-type": "application/json; charset=utf-8", "cache-control": "no-store" },
  });

/** Reads a small JSON body, refusing anything larger than the limit while streaming. */
async function readSmallJson(request: Request, limit: number): Promise<unknown> {
  const reader = request.body?.getReader();
  if (!reader) return null;
  const chunks: Uint8Array[] = [];
  let total = 0;
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    total += value.byteLength;
    if (total > limit) {
      await reader.cancel();
      throw new Error("too large");
    }
    chunks.push(value);
  }
  const bytes = new Uint8Array(total);
  let offset = 0;
  for (const chunk of chunks) {
    bytes.set(chunk, offset);
    offset += chunk.byteLength;
  }
  return JSON.parse(new TextDecoder().decode(bytes));
}

// KV values are cached at the edge for this long (KV minimum); publish/rollback propagate within it.
const KV_CACHE_TTL = 30;

function page(status: number, message: string, headers: Record<string, string> = {}) {
  return new Response(
    `<!doctype html><html lang="ko"><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1"><meta name="robots" content="noindex"><title>${message}</title><body style="font-family:system-ui;padding:48px"><p>${message}</p></body></html>`,
    { status, headers: { "content-type": "text/html; charset=utf-8", "cache-control": "no-store", ...headers } },
  );
}

async function serveObject(env: Env, key: string, cacheControl: string) {
  const object = await env.SITES.get(key);
  if (!object) return null;
  const headers = new Headers();
  object.writeHttpMetadata(headers);
  headers.set("etag", object.httpEtag);
  headers.set("cache-control", cacheControl);
  return new Response(object.body, { headers });
}

/** One KV read for M2 records; M1 records (bare siteId) need a second read for the pointer. */
async function resolveHost(env: Env, hostname: string): Promise<HostRecord | null> {
  const raw = await env.ROUTES.get(kvHostKey(hostname), { cacheTtl: KV_CACHE_TTL });
  if (!raw) return null;
  if (raw.startsWith("{")) return JSON.parse(raw) as HostRecord;
  const pointer = await env.ROUTES.get<PublishedPointer>(kvSiteKey(raw), { type: "json", cacheTtl: KV_CACHE_TTL });
  return { siteId: raw, status: "active", pointer, redirectTo: null };
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);
    const hostname = url.hostname.toLowerCase();

    // Visitor inquiry: the site is identified by Host here, then handed to the api by RPC.
    if (url.pathname === "/api/inquiry") {
      if (request.method !== "POST") return jsonResponse(405, { ok: false, error: "method not allowed" });
      const host = await resolveHost(env, hostname);
      if (!host || host.status !== "active" || !host.pointer) {
        return jsonResponse(404, { ok: false, error: "문의를 받을 수 없는 사이트입니다" });
      }
      if (!(request.headers.get("content-type") ?? "").startsWith("application/json")) {
        return jsonResponse(415, { ok: false, error: "JSON으로 보내 주세요" });
      }
      let body: unknown;
      try {
        body = await readSmallJson(request, MAX_FORM_BYTES);
      } catch {
        return jsonResponse(400, { ok: false, error: "요청을 읽지 못했습니다" });
      }
      const result = await env.API.submitInquiry({
        siteId: host.siteId,
        hostname,
        ip: request.headers.get("cf-connecting-ip"),
        body,
      });
      return jsonResponse(result.status, result.body);
    }

    if (request.method !== "GET" && request.method !== "HEAD") {
      return new Response("method not allowed", { status: 405 });
    }

    // Shared assets are content-addressed by version, so any host may serve them.
    if (url.pathname.startsWith("/_shared/")) {
      return (
        (await serveObject(env, r2SharedKey(url.pathname.slice("/_shared/".length)), "public, max-age=31536000, immutable")) ??
        new Response("not found", { status: 404 })
      );
    }
    // Uploaded photos are public and content-addressed.
    const media = MEDIA_PATH.exec(url.pathname);
    if (media) {
      return (
        (await serveObject(env, r2MediaKey(media[1]!, media[2]!), "public, max-age=31536000, immutable")) ??
        new Response("not found", { status: 404 })
      );
    }

    const host = await resolveHost(env, hostname);
    if (!host) return page(404, "연결된 사이트가 없습니다.");
    if (host.status === "cancelled") return page(410, "운영이 종료된 사이트입니다.");
    if (host.status === "suspended") return page(503, "잠시 운영을 멈춘 사이트입니다.", { "retry-after": "86400" });
    if (host.redirectTo) {
      return Response.redirect(`${url.protocol}//${host.redirectTo}${url.pathname}${url.search}`, 301);
    }
    const pointer = host.pointer;
    if (!pointer) return page(404, "아직 공개되지 않은 사이트입니다.");

    if (url.pathname === "/robots.txt") {
      return new Response(`User-agent: *\nAllow: /\nSitemap: https://${hostname}/sitemap.xml\n`, {
        headers: { "content-type": "text/plain; charset=utf-8", "cache-control": "public, max-age=3600" },
      });
    }
    const pages = pointer.pages ?? ["/"];
    if (url.pathname === "/sitemap.xml") {
      const lastmod = pointer.publishedAt.slice(0, 10);
      const urls = pages
        .map((route) => `<url><loc>https://${hostname}${route}</loc><lastmod>${lastmod}</lastmod></url>`)
        .join("");
      return new Response(
        `<?xml version="1.0" encoding="UTF-8"?>\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">${urls}</urlset>\n`,
        { headers: { "content-type": "application/xml; charset=utf-8", "cache-control": "public, max-age=3600" } },
      );
    }
    const route = url.pathname === "/index.html" ? "/" : url.pathname.replace(/\/(index\.html)?$/, "") || "/";
    if (pages.includes(route)) {
      const path = route === "/" ? "index.html" : `${route.slice(1)}/index.html`;
      const response = await serveObject(env, r2PageKey(host.siteId, pointer.versionId, path), "public, max-age=60");
      if (response) {
        response.headers.set("x-site-version", String(pointer.number));
        return response;
      }
      return page(404, "페이지 파일을 찾지 못했습니다.");
    }
    return page(404, "없는 주소입니다.");
  },
};
